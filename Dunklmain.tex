\begin{definition}
Let \(1\le r<\infty\). For a measurable set \(E\subset\mathbb R^N\) of finite positive measure and \(f\in L^r(E)\), define
\[
\operatorname{osc}_r(f;E)
:=
\left(
\frac1{|E|}
\int_E|f-(f)_E|^r\,dx
\right)^{1/r}.
\]
\end{definition}
\begin{definition}
Let \(0<p<\infty\), \(0<q\leq\infty\), and \(1\leq r<\infty\). Define
\[
OSC^{p,q}(\mathbb R^N)
:=
\left\{
f\in L^r_{\mathrm{loc}}(\mathbb R^N):
\|f\|_{OSC^{p,q}(\mathbb R^N)}<\infty
\right\},
\]
where
\[
\|f\|_{OSC^{p,q}(\mathbb R^N)}
:=
\sum_{\nu=1}^{2^N}
\left\|
\left\{
\operatorname{osc}_r(f;Q)
\right\}_{Q\in\mathbb D^\nu}
\right\|_{\ell^{p,q}}.
\]
\end{definition}

\subsection{Harmonic analysis in the Dunkl setting}
Let $\mathbb{R}^N$ be the Euclidean space equipped with the usual scalar product $\langle \cdot,\,\cdot\rangle $ and the induced norm $\|\cdot\|$. For a nonzero vector $v\in \mathbb{R}^N$, the reflection $\sigma_v$ with respect to the hyperplane $v^{\perp}$ orthogonal to $v$ is defined by
\begin{align*}
	\sigma_v x:= x-2\frac{\langle x,v\rangle}{\|v\|^2}v.
\end{align*}
A finite set $R\subset \mathbb{R}^N\backslash \{0\}$ is called a root system if $\sigma_v(R)=R$ for every $v\in R$. A root system $R$ is called a normalized reduced root system if  $\|v\|^2=2$ for every $v\in R$.
The finite group $G$ generated by the reflections $\{\sigma_v\}_{v\in R}$ is called the Weyl group (reflection group) of the root system $R$.
Let $k: R \to [0,\infty)$ be a multiplicity function, which is a $G$-invariant function  fixed throughout the paper. We will denote by $\mathcal O(x)=\{\sigma(x): \sigma \in G\}$ the $G$-orbit of a point $x\in\mathbb{R}^N$, and $\mathcal O(E)=\bigcup_{x \in E} \mathcal O(x)$ the $G$-orbit of $E\subset \mathbb{R}^N$.

Define the Dunkl measure $dw$ on $\mathbb{R}^N$ by
$$
dw(x) :=\prod_{v \in R} |\langle x,v\rangle|^{k(v)} dx,
$$
where $dx$ stands for the Lebesgue measure on $\mathbb{R}^N$. Let $B(x,r):=\{y\in \mathbb{R}^N:\|x-y\|< r\}$ be the Euclidean ball centred at $x$ with radius $r>0$. Let $\N:=N+\sum_{v\in R}k(v)$. Note that
$$w(B(tx,tr))=t^{\N}w(B(x,r)),\ {\rm for}\ x\in\mathbb{R}^N,\ t,r>0.$$
Furthermore,
\begin{align}\label{twosideinequ}
  w(B(x,r))\sim r^N \prod_{v \in R}(|\langle x, v \rangle |+r)^{k(v)},
\end{align}
where the implicit constant only depends on $N,R$ and $k$.  It follows from \eqref{twosideinequ} that there exists a constant $C\geq 1$ such that for every $x\in \mathbb{R}^N$,
\begin{align}\label{doublinginequality}
C^{-1}\left(\frac{r_2}{r_1}\right)^N\leq \frac{w(B(x,r_2))}{w(B(x,r_1))}\leq C\left(\frac{r_2}{r_1}\right)^\N,\ {\rm for}\ 0<r_1<r_2,
\end{align}
so the numbers $\N$ and $N$ are called the upper and lower homogeneous dimensions, respectively.

\begin{lemma}\label{lem:Dunkl-weight-A-infty}
Let
\[
dw(x)=\varpi(x)\,dx,
\qquad
\varpi(x)=\prod_{v\in R_+}|\langle x,v\rangle|^{2k(v)}.
\]
Then $\varpi\in A_\infty(\mathbb R^N,dx)$.
\end{lemma}
\begin{proof}
The assertion is immediate if $k\equiv0$. Otherwise, choose $p>1$ sufficiently large so that
\[
\frac{2|R_+|k(v)}{p-1}<1,
\qquad v\in R_+.
\]
For every $0<\beta<1$, $v\in R_+$, and Euclidean ball $B(x_0,r)$, we apply an orthogonal change of variables to deduce that
\[
\begin{aligned}
\fint_{B(x_0,r)}
|\langle x,v\rangle|^{-\beta}\,dx
&\lesssim
\frac{1}{r|v|}
\int_{\langle x_0,v\rangle-|v|r}^{\langle x_0,v\rangle+|v|r}
|s|^{-\beta}\,ds\\
&=
(|v|r)^{-\beta}
\int_{\frac{\langle x_0,v\rangle}{|v|r}-1}^{\frac{\langle x_0,v\rangle}{|v|r}+1}
|s|^{-\beta}\,ds\\
&\lesssim
(|v|r)^{-\beta}
\left(
1+\frac{|\langle x_0,v\rangle|}{|v|r}
\right)^{-\beta}\\
&\lesssim
\bigl(|\langle x_0,v\rangle|+r\bigr)^{-\beta}.
\end{aligned}
\]
Hence, by H\"older's inequality,
\[
\begin{aligned}
\fint_{B(x_0,r)}
\varpi^{-\frac1{p-1}}\,dx
&=
\fint_{B(x_0,r)}
\prod_{v\in R_+}
|\langle x,v\rangle|^{-\frac{2k(v)}{p-1}}\,dx\\
&\leq
\prod_{v\in R_+}
\left(
\fint_{B(x_0,r)}
|\langle x,v\rangle|^{-\frac{2|R_+|k(v)}{p-1}}\,dx
\right)^{\frac1{|R_+|}}\\
&\lesssim
\prod_{v\in R_+}
\left(
|\langle x_0,v\rangle|+r
\right)^{-\frac{2k(v)}{p-1}}.
\end{aligned}
\]
On the other hand, by \eqref{twosideinequ} and the
$G$-invariance of $k$,
\[
\begin{aligned}
\fint_{B(x_0,r)}\varpi\,dx
=
\frac{w(B(x_0,r))}{|B(x_0,r)|}
\lesssim
\prod_{v\in R_+}
\bigl(|\langle x_0,v\rangle|+r\bigr)^{2k(v)}.
\end{aligned}
\]
Therefore,
\[
\left(\fint_B\varpi\,dx\right)
\left(\fint_B\varpi^{-\frac1{p-1}}\,dx\right)^{p-1}
\lesssim1
\]
uniformly over all Euclidean balls $B$. Thus $\varpi\in A_p(\mathbb R^N,dx)$, and hence $\varpi\in A_\infty(\mathbb R^N,dx)$.
\end{proof}

The Dunkl operators $T_\xi$ are defined by
$$T_\xi f(x):=\partial_\xi f(x)+\sum_{v\in R}\frac{k(v)}{2}\langle v,\xi\rangle\frac{f(x)-f(\sigma_v(x))}{\langle v,x\rangle},$$
where $\partial_\xi f$ denotes the directional derivative of $f$ in the direction $\xi$. Furthermore, the Dunkl Laplacian is defined by $$\Delta_{\D}f(x):=\sum_{j=1}^N T_{e_j}^2f(x)=\Delta_{\mathbb{R}^N}f(x)+\sum_{v\in R}k(v)\delta_v f(x),$$
where $\Delta_{\mathbb{R}^N}$ is the Laplacian on $\mathbb{R}^N$ and $\delta_v$ is defined by
$$\delta_v f(x):=\frac{\partial_v f(x)}{\langle v,x\rangle}-\frac{f(x)-f(\sigma_v(x))}{\langle v,x\rangle^2}.$$
In this article, we consider the Dunkl-Riesz transforms defined by $R_{\D,j}:=T_{e_j}(-\Delta_{\D})^{-1/2}$, $j=1,2,\ldots,N$.


\begin{definition}{\rm
The Weyl chambers are defined as the closures of connected components of the set
  \begin{align}
    \left\{x\in\mathbb{R}^N:\langle x,v\rangle \neq 0\ \ { for} \ { all} \ v\in R\right\}.\end{align}}
\end{definition}
Define $
	d(x,y):=\min_{\sigma\in G}\|x-\sigma(y)\|
$
as the distance between two $G$-orbits $\mathcal O(x)$ and $\mathcal O(y)$. Recall from \cite[Chapter VII, proof of Theorem 2.12]{MR514561} that
\begin{align}\label{distanceequa}
d(x,y)=\|x-y\|\ {\rm if}\ {\rm and}\ {\rm only}\ {\rm if}\ x,y\in\Gamma
\end{align}
for some closed Weyl chamber $\Gamma$.

It is clear that $\mathcal O(B(x,r)) = \{y\in \mathbb{R}^N: d(x,y)<r\}$. Moreover,
\[
w(B(x,r)) \leq w(\mathcal O(B(x,r))) \leq |G|w(B(x,r))
\]
for all $x \in \mathbb{R}^N$ and $r>0$. We also denote $B^d(x,r):= \{y\in \mathbb{R}^N: d(x,y)<r\}$, i.e., $B^d(x,r)=\mathcal O(B(x,r))$.

Denote by $K_{\D,j}(x,y)$ the integral kernel of $R_{\D,j}$. For every $x,y\in\mathbb{R}^N$ and $r>0$, let
$$V(x,y,r):=\max\{w(B(x,r)),w(B(y,r))\}.$$
The following lemma establishes derivative estimates for the Dunkl--Riesz transform kernel.
\begin{lemma}
\label{lem:Dunkl-Riesz-endpoint-derivatives}
Assume that $N\geq 1$. Then for every pair of multi-indices $\alpha,\beta$, every $j\in\{1,\ldots,N\}$, and every $x,y\in\mathbb R^N$ with $d(x,y)>0$,
\begin{equation}
\label{eq:Dunkl-Riesz-endpoint-derivatives}
\left|
\partial_x^\alpha\partial_y^\beta
K_{{\rm Dunkl},j}(x,y)
\right|
\lesssim_{\alpha,\beta,R,k}
\frac{d(x,y)}{\|x-y\|}
\frac{d(x,y)^{-(|\alpha|+|\beta|)}}
{V(x,y,d(x,y))}.
\end{equation}
\end{lemma}
\begin{proof}
{\bf Case 1.} Suppose that $|\alpha|+|\beta|=0$.
By \cite[Theorem~1.1]{MR4733961},
\[
|K_{{\rm Dunkl},j}(x,y)|
\lesssim
\frac{d(x,y)}{\|x-y\|}
\frac{1}{w(B(x,d(x,y)))}.
\]
By the definition of $d$, the $G$-invariance of $dw$, and
\eqref{doublinginequality},
\[
V(x,y,d(x,y))
\lesssim
w(B(x,d(x,y))),
\]
and hence \eqref{eq:Dunkl-Riesz-endpoint-derivatives} follows.

{\bf Case 2.} Suppose that $|\alpha|+|\beta|\geq1$. By \cite[Theorem~3.1]{MR4042416},
\[
\left|
\partial_x^\alpha\partial_y^\beta h_t(x,y)
\right|
\lesssim_{\alpha,\beta,R,k}
t^{-\frac{|\alpha|+|\beta|}{2}}
\left(1+\frac{\|x-y\|}{\sqrt t}\right)^{-2}
V(x,y,\sqrt t)^{-1}
e^{-cd(x,y)^2/t}.
\]
Recall from the proof of Lemma~\ref{nondegenerate} that
\[
K_{{\rm Dunkl},j}(x,y)
=
-\frac{C}{2}\int_0^\infty
(y_j-x_j)h_t(x,y)\,\frac{dt}{t^{3/2}}.
\]
Since
\[
\partial_x^\alpha\partial_y^\beta
\bigl((y_j-x_j)h_t(x,y)\bigr)
=
(y_j-x_j)\partial_x^\alpha\partial_y^\beta h_t(x,y)
-\alpha_j\partial_x^{\alpha-e_j}\partial_y^\beta h_t(x,y)
+\beta_j\partial_x^\alpha\partial_y^{\beta-e_j}h_t(x,y),
\]
where the corresponding term is omitted if $\alpha_j=0$ or $\beta_j=0$,
we obtain
\begin{align*}
\left|
\partial_x^\alpha\partial_y^\beta
K_{{\rm Dunkl},j}(x,y)
\right|
&\lesssim
\frac1{\|x-y\|}
\int_0^\infty
t^{-\frac{|\alpha|+|\beta|+1}{2}}
V(x,y,\sqrt t)^{-1}e^{-cd(x,y)^2/t}\,dt \\
&\quad+
\frac1{\|x-y\|^2}
\int_0^\infty
t^{-\frac{|\alpha|+|\beta|}{2}}
V(x,y,\sqrt t)^{-1}e^{-cd(x,y)^2/t}\,dt.
\end{align*}
By \eqref{doublinginequality},
\[
V(x,y,\sqrt t)^{-1}
\lesssim
V(x,y,d(x,y))^{-1}
\begin{cases}
\displaystyle \left(\frac{d(x,y)}{\sqrt t}\right)^{\N},
&0<t\leq d(x,y)^2,\\[2mm]
\displaystyle \left(\frac{d(x,y)}{\sqrt t}\right)^N,
&t\geq d(x,y)^2.
\end{cases}
\]
Hence,
\begin{align*}
&\int_0^\infty
t^{-\frac{|\alpha|+|\beta|+1}{2}}
V(x,y,\sqrt t)^{-1}e^{-cd(x,y)^2/t}\,dt\\
&\quad\lesssim
\frac1{V(x,y,d(x,y))}
\Bigg(
d(x,y)^\N
\int_0^{d(x,y)^2}
t^{-\frac{\N+|\alpha|+|\beta|+1}{2}}
e^{-cd(x,y)^2/t}\,dt
+
d(x,y)^N
\int_{d(x,y)^2}^\infty
t^{-\frac{N+|\alpha|+|\beta|+1}{2}}\,dt
\Bigg)\\
&\quad\lesssim
\frac{d(x,y)^{1-|\alpha|-|\beta|}}
{V(x,y,d(x,y))}.
\end{align*}
The first integral in the parentheses is estimated by the change of variables
$s=d(x,y)^2/t$, while the second one converges since
$N+|\alpha|+|\beta|>1$.

{\bf Case 2.1.} Suppose that $N+|\alpha|+|\beta|>2$. We apply the same argument to see that
\begin{align*}
&\int_0^\infty
t^{-\frac{|\alpha|+|\beta|}{2}}
V(x,y,\sqrt t)^{-1}e^{-cd(x,y)^2/t}\,dt\\
&\quad\lesssim
\frac1{V(x,y,d(x,y))}
\Bigg(
d(x,y)^\N
\int_0^{d(x,y)^2}
t^{-\frac{\N+|\alpha|+|\beta|}{2}}
e^{-cd(x,y)^2/t}\,dt
+
d(x,y)^N
\int_{d(x,y)^2}^\infty
t^{-\frac{N+|\alpha|+|\beta|}{2}}\,dt
\Bigg)\\
&\quad\lesssim
\frac{d(x,y)^{2-|\alpha|-|\beta|}}
{V(x,y,d(x,y))}.
\end{align*}
Since $d(x,y)\leq\|x-y\|$, it follows that
\[
\left|
\partial_x^\alpha\partial_y^\beta
K_{{\rm Dunkl},j}(x,y)
\right|
\lesssim_{\alpha,\beta,R,k}
\frac{d(x,y)}{\|x-y\|}
\frac{d(x,y)^{-(|\alpha|+|\beta|)}}
{V(x,y,d(x,y))}.
\]

{\bf Case 2.2.} Suppose that $N=1$ and $|\alpha|+|\beta|=1$. For the terms in which a derivative falls on $y_j-x_j$, retaining the factor
$\bigl(1+\|x-y\|/\sqrt t\bigr)^{-2}$ and splitting the integral at
$d(x,y)^2$ and $\|x-y\|^2$, we obtain from \eqref{doublinginequality}
\begin{align*}
&\int_0^\infty
t^{-3/2}
\left(1+\frac{\|x-y\|}{\sqrt t}\right)^{-2}
V(x,y,\sqrt t)^{-1}
e^{-cd(x,y)^2/t}\,dt\\
&\quad\lesssim
\frac{d(x,y)^\N}{\|x-y\|^2V(x,y,d(x,y))}
\int_0^{d(x,y)^2}
t^{-\frac{\N+1}{2}}e^{-cd(x,y)^2/t}\,dt\\
&\qquad+
\frac{d(x,y)}{\|x-y\|^2V(x,y,d(x,y))}
\int_{d(x,y)^2}^{\|x-y\|^2}\frac{dt}{t}
+
\frac{d(x,y)}{V(x,y,d(x,y))}
\int_{\|x-y\|^2}^\infty\frac{dt}{t^2}\\
&\quad\lesssim
\frac{1}{\|x-y\|V(x,y,d(x,y))},
\end{align*}
where we used
\[
\frac{d(x,y)}{\|x-y\|}
\log\frac{\|x-y\|}{d(x,y)}
\lesssim 1.
\]
Combining this estimate with the preceding estimate for the first term, we obtain
\[
\left|
\partial_x^\alpha\partial_y^\beta
K_{{\rm Dunkl},j}(x,y)
\right|
\lesssim_{\alpha,\beta,R,k}
\frac{1}{\|x-y\|V(x,y,d(x,y))}
=
\frac{d(x,y)}{\|x-y\|}
\frac{d(x,y)^{-1}}
{V(x,y,d(x,y))}.
\]
This completes the proof of Lemma~\ref{lem:Dunkl-Riesz-endpoint-derivatives}.
\end{proof}

\begin{lemma}\label{nondegenerate}
For $j\in\{1,2,\ldots,N\}$, and for every ball $B=B(x_0,r)\subset \mathbb{R}^N$, there is another ball $\widehat{B}=B(y_0,r)$ with $y_0=x_0+5re_j$ such that for every $(x,y)\in B\times \widehat{B}$, $K_{\D,j}(x,y)$ does not change sign and satisfies
\begin{align*}
|K_{\D,j}(x,y)|\geq\frac{C}{w(B(x_0,r))}.
\end{align*}
\end{lemma}
\begin{proof}
The lower bound is given in \cite[Theorem 1.2]{MR4733961}, while the fact that $K_{\D,j}(x,y)$ does not change sign can be seen in the proof of \cite[Theorem 1.2]{MR4733961}. Indeed, with the choice $y_0:=x_0+5re_j$, we have $y_j-x_j\geq 3r$ for all $(x,y)\in B\times \widehat{B}$. Together with the fact that the heat kernel $h_t(x,y)$ associated with the Dunkl Laplacian is  positive, the representation
$$K_{\D,j}(x,y)=-\frac{C}{2}\int_0^\infty \frac{y_j-x_j}{t}h_t(x,y)\frac{dt}{\sqrt{t}},$$
implies that $K_{\D,j}(x,y)<0$ (and thus, does not change sign) on $ B\times \widehat{B}$.
\end{proof}



\subsection{Systems of dyadic cubes}


\begin{definition}\label{Defsystem}{\rm
Let $X\subseteq \mathbb{R}^N$ be a nonempty open set. A countable family
$
    \mathscr{D}
    := \bigcup_{k\in\mathbb{Z}}\mathscr{D}_{k}, \
    \mathscr{D}_{k}
    :=\{Q^k_{\alpha}\colon \alpha\in \mathscr{A}_k\},
$
of Borel sets $Q^k_{\alpha}\subseteq X$ is called \textit{a
system of dyadic cubes over  $X$}  if it has the following properties:

\begin{enumerate}
  \item[(i)] For each $k\in\mathbb{Z}$, we have $$    X
    = \bigcup_{\alpha\in \mathscr{A}_k} Q^k_{\alpha};$$\smallskip

  \item[(ii)] $\text{If }\ell\geq k\text{, then either }
        Q^{\ell}_{\beta}\subseteq Q^k_{\alpha}\text{ or }
        Q^k_{\alpha}\cap Q^{\ell}_{\beta}=\varnothing$;\smallskip

  \item[(iii)] $
    \text{For each }(k,\alpha)\text{ and each } \ell\leq k,
    \text{ there exists a unique } \beta
    \text{ such that }Q^{k}_{\alpha}\subseteq Q^\ell_{\beta};
$

\smallskip

\item[(iv)] For each $(k,\alpha)$ there exist $2^N$ indices $\beta$ such that
    $$ Q^{k+1}_{\beta}\subseteq Q^k_{\alpha},\ {\rm and}\
        Q^k_{\alpha} =\bigcup_{{\substack{Q\in\mathscr{D}_{k+1}\\ Q\subseteq Q^k_{\alpha}}}}Q;$$

\smallskip

\item[(v)] For each $(k,\alpha)$, one has $$B_X(x^k_{\alpha},A_1 2^{-k})
    \subseteq Q^k_{\alpha}\subseteq B_X(x^k_{\alpha},A_2 2^{-k})
    =: B(Q^k_{\alpha})$$
    for some positive constants $A_1,A_2$ depending only on the geometry of $X$, {where we set $B_X(x,r):=B(x,r)\cap X$ for all $x\in X$ and $r>0$.}

\smallskip

\item [(vi)]  If $\ell\geq k$ and
   $Q^{\ell}_{\beta}\subseteq Q^k_{\alpha}$, then
   $$B(Q^{\ell}_{\beta})\subseteq B(Q^k_{\alpha}).$$
\end{enumerate}



   }
\end{definition}

Each $Q^k_{\alpha}$ is called a \textit{dyadic cube of generation} $k$, with center $x^k_{\alpha}\in Q^k_{\alpha}$ and side-length $\ell(Q^k_{\alpha})=2^{-k}$.
\begin{nota}
For every dyadic cube $Q\in\mathscr{D}$, we denote by $c_Q$ and $\ell(Q)$ its center and side-length, respectively.
\end{nota}

\begin{nota}\label{standardsystem}
Let $$\mathbb{D}^0:=\bigcup_{k\in\mathbb{Z}}\mathbb{D}_k^0$$ be the standard system of dyadic cubes over $\mathbb{R}^N$.
Here each dyadic cube is a product of half-open intervals
\[
[m_1 2^{-k}, (m_1+1)2^{-k}) \times \cdots \times [m_N 2^{-k}, (m_N+1)2^{-k}),
\]
with $(m_1,\ldots,m_N)\in \mathbb{Z}^N$. Moreover, we let $$\mathbb{D}_{\Gamma}:=\bigcup_{k\in\mathbb{Z}}\mathbb{D}_{\Gamma,k}$$ be the system of dyadic cubes over a fixed chamber $\Gamma$, whose  construction will be given in Corollary \ref{Construction over gamma}.
\end{nota}


\begin{lemma}\label{Nsystems}
There exist $2^N$ systems of dyadic cubes $\mathbb{D}^1,\mathbb{D}^2,\ldots,\mathbb{D}^{2^N}$ such that for any ball $B=B(x_B,r_B)\subset \mathbb{R}^N$ with $\frac{2^{-k-2}}{3}\leq r_B< \frac{2^{-k-1}}{3}$ for some $k\in\mathbb{Z}$,  there exists a cube ${\displaystyle Q\in\bigcup_{\nu=1}^{2^N}\mathbb{D}^\nu_k}$ such that
$$B\subset Q\subset 12\sqrt{N} B.$$
\end{lemma}

\begin{proof}
{\bf $\bullet$ Step 1.}  {\it Recall the one-dimensional setting given in \cite[Section 2]{MR3420475}.}

Let $\mathbb{D}^{(0)}(\mathbb{R})$ be the standard system of dyadic intervals over $\mathbb{R}$ (see Notation \ref{standardsystem}).
Let $\delta=1/3$ and define
$$
   \mathbb D^{(\delta)}(\mathbb R)=\bigcup_{k\in\mathbb Z} \mathbb D^{(\delta)}_k(\mathbb R),
$$
where for $k\geq0$,
$$
    \mathbb D^{(\delta)}_k(\mathbb R)=\{I+\delta \mid I\in \mathbb D_k^{(0)}(\mathbb R)\},
$$
while for $k<0$ and $k$ even,
$$
    \mathbb D^{(\delta)}_k(\mathbb R)
    =\Big\{\Big[{m\over 2^k}+\delta+\sum_{j=(k/2)+1}^0 2^{-2j},\ {m+1\over 2^k}
    +\delta+\sum_{j=(k/2)+1}^0 2^{-2j}\Big)\ \Big|\ m\in
    \mathbb Z\Big\}.
$$
These choices together with the nestedness property property completely
determine the collections $\mathbb D^{(\delta)}_k(\mathbb R)$ for $k<0$, $k$ odd.

Fix an interval $J\subset\mathbb R$ with $2^{-k-1}/3\le |J|<2^{-k}/3$ for some $k\in\mathbb Z$. Now we set
$A_k=\{ m\cdot 2^{-k} \mid m\in\mathbb Z \}$ for every fixed $k$ and
\begin{enumerate}
    \item[(1)]$A_k^{(\delta)}=\{ \delta+ m\cdot 2^{-k} \mid m\in\mathbb Z
        \}$ for $k\geq 0$,

    \item[(2)]$A_k^{(\delta)}=\{ \delta+
        \sum_{j=(k/2)+1}^{0}2^{-2j}+ m\cdot 2^{-k} \mid m\in\mathbb Z
        \}$ for $k<0$, $k$ even, and

    \item[(3)]$A_k^{(\delta)}=\{ \delta+
        \sum_{j=(k-1)/2+1}^{0}2^{-2j}+ m\cdot 2^{-k} \mid
        m\in\mathbb Z \}$ for $k<0$, $k$ odd.
\end{enumerate}
Note that for any two different points $a,b\in A_k\cup
A_k^{(\delta)}$, we have $|a-b|\geq 2^{-k}/3>|J|$. Thus,
there is at most one element of $A_k\cup A_k^{(\delta)}$ contained in
$J$. Hence, $J\cap A_k=\emptyset $ or $J\cap
A_k^{(\delta)}=\emptyset $. Therefore, $J$ must be contained in
some dyadic interval $I\in \mathbb D_k^{(0)}$ or $I\in \mathbb D^{(\delta)}_k$ and
$|I|=2^{-k}\leq 6 |J|$.

{\bf $\bullet$ Step 2.} {\it We extend the construction to $\mathbb{R}^N$ ($N\geq 1$).}

For $\vartheta=(\vartheta_1,\vartheta_2,\ldots,\vartheta_N)\in\{0,\delta\}^N$, define
$$\mathbb{D}^{(\vartheta)}:=\bigcup_{k\in\mathbb{Z}}\mathbb{D}_k^{(\vartheta)},\quad \mathbb{D}_k^{(\vartheta)}:=\left\{\prod_{j=1}^N I_j: I_j\in \mathbb{D}_k^{(\vartheta_j)}\right\}.$$
Applying Step 1 to $\pi_i(B)$, where $\pi_i(B)$ denotes the $i$-th coordinate projection, we obtain $I_i\in\mathbb{D}_k^{(0)}$ or $I_i\in\mathbb{D}_k^{(\delta)}$ with $\pi_i(B)\subset I_i$ and $|I_i|=2^{-k}\leq 12 r_B$. Let $\vartheta_i\in\{0,\delta\}$
be the corresponding shift so that $I_i\in\mathbb D_k^{(\vartheta_i)}$, set $\vartheta:=(\vartheta_1,\ldots,\vartheta_N)$, and let $Q=\prod_{j=1}^NI_j\in\mathbb{D}_k^{(\vartheta)}$. It follows that $B\subset Q$ and $\ell(Q)=2^{-k}\leq 12r_B$. Hence, for every $y\in Q$,
$$\|y-x_B\|\leq \sqrt{N}\ell(Q)\leq 12\sqrt{N}r_B.$$
This implies that $Q\subset B(x_B,12\sqrt{N}r_B)$. Enumerating $\big\{\mathbb D^{(\vartheta)}:\vartheta\in\{0,\delta\}^N\big\}$ as
$\mathbb D^1,\ldots,\mathbb D^{2^N}$, we complete the proof of Lemma \ref{Nsystems}.
\end{proof}

\end{enumerate}
}

{\bf Convention. }
Throughout this paper, we will use the following notation:
\begin{itemize}
  \item Let $R
  \subset \mathbb{R}^{N}$ be a finite normalized reduced root system;
  \item Let
$W:=\operatorname{span}(R)$ with $N_{0}:=\dim W$;
\item Let
$
  \Omega:=\mathbb{R}_+^{N_{0}}\times \mathbb{R}^{N-N_{0}};
$
\item Let $\Gamma_+$ be a fixed open Weyl chamber (see Definition \ref{def:weyl_chamber_positive_roots}) and $\Gamma$ be its closure.
\end{itemize}


\begin{definition}{\rm \cite[Definition ~2.7]{HeckmanCoxeterGroups}}\label{def:weyl_chamber_positive_roots} A connected component of \begin{align}
    \left\{x\in\mathbb{R}^N:\langle x,v\rangle \neq 0\ \ { for} \ { all} \ v\in R\right\}
  \end{align} is called an open
Weyl chamber. Its closure is called a Weyl chamber.  Define the corresponding sets of \emph{positive} and \emph{negative} roots by
\[
R_{+}:=\{\alpha\in R:\ \langle x,\alpha\rangle>0\ \text{for all }x\in \Gamma_{+}\},
\qquad
R_{-}:=-R_{+}.
\]
\end{definition}

\begin{definition}{\rm \cite[Definition~2.9]{HeckmanCoxeterGroups}}\label{def:simple_roots}
A root $\alpha\in R_{+}$
is called simple in $R_{+}$ if it cannot be written as
\[
\alpha=\lambda_{1}\alpha_{1}+\lambda_{2}\alpha_{2}
\quad\text{with }\lambda_{1},\lambda_{2}\ge 1\ \text{and}\ \alpha_{1},\alpha_{2}\in R_{+}.
\]
The set $\Delta\subset R_{+}$ of all simple roots is called the
simple system.
\end{definition}

\begin{lemma}\label{lem:simple_system_associated_to_chamber}
With the notation given in Definition \ref{def:simple_roots},  $\Delta$ is a basis of $W$. If we denote by
$
\Delta=\{\alpha_{1},\alpha_{2},\ldots,\alpha_{N_{0}}\},
$
then the open and closed chambers can be written as
\begin{align}
\Gamma_{+}
&=\Bigl\{x\in\mathbb{R}^{N}:\ \langle x,\alpha_{i}\rangle>0,\ i=1,\ldots,N_{0}\Bigr\},
\label{eq:open_chamber_by_simple_roots}\\
\Gamma
&=\Bigl\{x\in\mathbb{R}^{N}:\ \langle x,\alpha_{i}\rangle\ge 0,\ i=1,\ldots,N_{0}\Bigr\},
\label{eq:closed_chamber_by_simple_roots}
\end{align}
and every root $\alpha\in R$ admits a unique expansion
\[
\alpha=\sum_{i=1}^{N_{0}}c_{i}\alpha_{i}
\]
such that either $c_{i}\ge 0$ for all $i$ (when $\alpha\in R_{+}$), or $c_{i}\le 0$ for all $i$
(when $\alpha\in R_{-}$). In particular, the simple system $\Delta$ is uniquely determined by $\Gamma_{+}$
(equivalently, by $\Gamma$).
\end{lemma}

\begin{proof}
%We regard $R$ as a normalized root system in the Euclidean space $W$ endowed with the
%restricted inner product. Since $\sigma_{\alpha}(\alpha)=-\alpha$ and $\sigma_{\alpha}(R)=R$
%for every $\alpha\in R$, we have $R=-R$.

Let $\Delta=\{\alpha_{1},\ldots,\alpha_{n}\}\subset R_{+}$ be the set of simple roots in $R_{+}$. By \cite[Proposition~2.10]{HeckmanCoxeterGroups}, every $\alpha\in R_{+}$
is a nonnegative linear combination of $\alpha_{1},\ldots,\alpha_{n}$. Since $R=R_{+}\cup R_{-}$ (this follows from the fact that for each $\alpha\in R$, the continuous function
$x\mapsto\langle x,\alpha\rangle$ does not vanish on the connected set $\Gamma_{+}$) and $R_{-}=-R_{+}$, it follows that
$R\subset\operatorname{span}(\Delta)$ and hence,
\[
W=\operatorname{span}(R)\subset\operatorname{span}(\Delta).
\]
The reverse inclusion $\operatorname{span}(\Delta)\subset W$ is immediate from $\Delta\subset R\subset W$,
so $\operatorname{span}(\Delta)=W$. On the other hand, it follows from \cite[Theorem~2.16]{HeckmanCoxeterGroups} that the simple roots are linearly
independent. Consequently, $\Delta$ is a basis
of $W$, and therefore $n=\dim W=N_{0}$. Moreover, every $\alpha\in R$ has a unique expansion
$\alpha=\sum_{i=1}^{N_{0}}c_{i}\alpha_{i}$. If $\alpha\in R_{+}$, the coefficients can be chosen
nonnegative by \cite[Proposition~2.10]{HeckmanCoxeterGroups}. If $\alpha\in R_{-}=-R_{+}$, then
$-\alpha\in R_{+}$ has an expansion with all coefficients nonnegative, and multiplying by $-1$ gives
an expansion of $\alpha$ with all coefficients nonpositive.
The descriptions \eqref{eq:open_chamber_by_simple_roots} and \eqref{eq:closed_chamber_by_simple_roots} follow from \cite[Corollary~2.11]{HeckmanCoxeterGroups}.


To see that $\Delta$ is uniquely determined by $\Gamma_+$, note that $\Gamma_+$ determines $R_{+}$
by Definition~\ref{def:weyl_chamber_positive_roots}, and the set of simple roots in $R_{+}$ is
characterized intrinsically by Definition~\ref{def:simple_roots}. Finally, since $\Gamma_+$ is the (unique) interior component of $\Gamma$, specifying $\Gamma$ determines $\Gamma_+$ and vice versa. This completes the proof of Lemma \ref{lem:simple_system_associated_to_chamber}.
\end{proof}
\begin{nota}
With the notation given in Lemma \ref{lem:simple_system_associated_to_chamber}, denote by
 $$\Delta=\{\alpha_{1},\ldots,\alpha_{N_{0}}\}$$ the simple system associated with $\Gamma$.
\end{nota}

\begin{lemma}\label{chambergeometry}
Each Weyl chamber $\Gamma$ admits the decomposition
\[
\Gamma = K(\Gamma)\times W^\perp,
\qquad
K(\Gamma):=\{w\in W:\ \langle w,\alpha\rangle\ge 0\ \text{for all }\alpha\in\Delta\}.
\]
\end{lemma}

\begin{proof}
Write $x=w+z$ with $w\in W$ and $z\in W^{\perp}$. Since $\alpha_{i}\in W$, we have $\langle x,\alpha_{i}\rangle=\langle w,\alpha_{i}\rangle$ for all
$i=1,\ldots,N_0$. This, together with \eqref{eq:closed_chamber_by_simple_roots}, implies that $x\in \Gamma$ if and only if
$
w\in K(\Gamma).
$
This implies that
$
\Gamma
=
K(\Gamma)\times W^{\perp},
$
and thus completes the proof.
\end{proof}

\begin{coro}\label{coro:chamber-orthant}
Let $\Phi_{W^{\perp}}$ be any linear isomorphism from $W^\perp$ to $\mathbb{R}^{N-N_{0}}$, and let
\[
\Phi_W:W\longrightarrow\mathbb{R}^{N_{0}},
\qquad
\Phi_W(w):=\big(\langle w,\alpha_1\rangle,\dots,\langle w,\alpha_{N_{0}}\rangle\big),
\]
\[
\Phi:\mathbb{R}^N=W\oplus W^\perp\longrightarrow
\mathbb{R}^{N_{0}}\oplus\mathbb{R}^{N-N_{0}}\cong\mathbb{R}^N,
\qquad
\Phi(w+z):=\big(\Phi_W(w),\,\Phi_{W^\perp}(z)\big).
\]
Then $\Phi$ is a linear isomorphism and
\[
\Phi(\Gamma)=\mathbb{R}_+^{N_{0}}\times \mathbb{R}^{N-N_{0}}.
\]
\end{coro}

\begin{proof}
By Lemma~\ref{chambergeometry} we have $\Gamma=K(\Gamma)\times W^\perp$, where
\[
K(\Gamma)=\{w\in W:\ \langle w,\alpha_i\rangle\ge0,\ i=1,\dots,N_{0}\}.
\]
Since $\Delta$ is a basis of $W$, the Gram matrix
$G=(\langle \alpha_i,\alpha_j\rangle)_{i,j=1}^{N_{0}}$ is nonsingular. Hence,
$\Phi_W:W\longrightarrow\mathbb{R}^{N_{0}}$ is a linear isomorphism. Moreover,
for $w\in W$ we have $w\in K(\Gamma)$ if and only if $\Phi_{W}(w)\in\mathbb{R}_+^{N_{0}}$,
so
\[
\Phi_{W}\big(K(\Gamma)\big)=\mathbb{R}_+^{N_{0}}.
\]
By construction, $\Phi$ is a linear isomorphism (as a direct sum of isomorphisms),
and therefore
\[
\Phi(\Gamma)
=\Phi\big(K(\Gamma)\times W^\perp\big)
=\Phi_{W}\big(K(\Gamma)\big)\times \Phi_{W^{\perp}}\big(W^\perp\big)
=\mathbb{R}_+^{N_{0}}\times \mathbb{R}^{N-N_{0}},
\]
which proves the desired conclusion.
\end{proof}


\begin{coro}\label{Construction over gamma}
 Let $\Gamma$ be an arbitrary Weyl chamber. There exists a system of dyadic cubes over $\Gamma$.
\end{coro}\begin{proof}
Let $\Phi:\mathbb{R}^N\to\mathbb{R}^N$ be the linear isomorphism from Corollary~\ref{coro:chamber-orthant} so that
\[
\Phi(\Gamma)=\Omega:=\mathbb{R}_+^{N_0}\times \mathbb{R}^{N-N_0}.
\]

For each $k\in\mathbb{Z}$, define $\mathbb{D}^0_{\Omega,k}$ to be the family of half-open cubes
\[
Q=\prod_{i=1}^{N} I_i,
\qquad
I_i=\big[m_i2^{-k},(m_i+1)2^{-k}\big),
\]
where $m_i\in\mathbb{Z}_+$ for $1\le i\le N_0$ and $m_i\in\mathbb{Z}$ for $N_0<i\le N$. Set
\[
\mathbb{D}^0_\Omega:=\bigcup_{k\in\mathbb{Z}}\mathbb{D}^0_{\Omega,k}.
\]
It is standard that $\mathbb{D}^0_\Omega$ is a system of dyadic cubes over $\Omega$.

Next, for each $k\in\mathbb{Z}$, define
\[
\mathbb{D}_{\Gamma,k}:=\big\{\Phi^{-1}(Q): Q\in \mathbb{D}^0_{\Omega,k}\big\},
\qquad
\mathbb{D}_{\Gamma}:=\bigcup_{k\in\mathbb{Z}}\mathbb{D}_{\Gamma,k}.
\]
Since $\Phi:\mathbb{R}^N\to\mathbb{R}^N$ is a linear isomorphism, we see that $\mathbb{D}_{\Gamma}$ is a system of dyadic cubes over $\Gamma$.
\end{proof}




\begin{nota}
 Let $\Phi$ be the linear isomorphism given in Corollary~\ref{coro:chamber-orthant}. For every Borel set $E\subset\Omega$, define
\begin{align}\label{defmu}
  \mu(E):=w\big(\Phi^{-1}(E)\big).
\end{align}
\end{nota}
\begin{lemma}\label{mulemma}
We have
  \begin{equation}\label{pushforward-density}
d\mu(u,z)=\phi(u)\,du\,dz,\quad \phi(u)=|{\rm det}(\Phi)|^{-1}\prod_{v\in R_+}\Big(\sum_{i=1}^{N_{0}} a_{v,i}\,u_i\Big)^{2k(v)}.
\end{equation}
where for each $v\in R_+$, the coefficients $a_{v,i}\ge0$ are uniquely determined by the simple-root expansion
\(v=\sum_{i=1}^{N_0} a_{v,i}\alpha_i\) and satisfy $(a_{v,1},\dots,a_{v,N_0})\neq\mathbf 0$.
\end{lemma}
\begin{proof}
Since $\Phi(\Gamma)=\Omega$, we have $\Phi^{-1}(\Omega)=\Gamma$. Hence, for every nonnegative Borel function
$f$ on $\Omega$,
\[
\int_{\Omega} f(u,z)\,d\mu(u,z)
=\int_{\Gamma} f\big(\Phi(x)\big)\,dw(x).
\]
We make the linear change of variables $(u,z)=\Phi(x)$, so that $x=\Phi^{-1}(u,z)$. Since $\Phi$ is linear and invertible, the Jacobian is constant and
\[
dx = |{\rm det}(\Phi)|^{-1}\,du\,dz.
\]
It follows from the definition of $\Phi$ that
$
u_i=\langle x,\alpha_i\rangle$ for $i=1,\dots,N_0.
$
This implies that
\begin{align}\label{changevariable}
\langle x,v\rangle
=\Big\langle x,\sum_{i=1}^{N_0} a_{v,i}\alpha_i\Big\rangle
=\sum_{i=1}^{N_0} a_{v,i}\langle x,\alpha_i\rangle
=\sum_{i=1}^{N_0} a_{v,i}u_i.
\end{align}
Substituting these identities into the previous integral, we obtain
\begin{align*}
\int_{\Omega} f(u,z)\,d\mu(u,z)
&=\int_{\Gamma} f\big(\Phi(x)\big)\Big(\prod_{v\in R_+}|\langle x,v\rangle|^{2k(v)}\Big)\,dx \\
&=\int_{\Omega} f(u,z)\,
|{\rm det}(\Phi)|^{-1}\prod_{v\in R_+}\Big(\sum_{i=1}^{N_0} a_{v,i}u_i\Big)^{2k(v)}\,du\,dz.
\end{align*}
Therefore, \eqref{pushforward-density} holds. This completes the proof of Lemma~\ref{mulemma}.
\end{proof}


\begin{lemma}\label{lem:gallery-simple}
Let $\Gamma_1$ and $\Gamma_2$ be two Weyl chambers. Then there exist an integer $M\ge 0$ and
$\alpha_1,\dots,\alpha_M\in R$ such that, if we set $\Gamma^{(0)}:=\Gamma_2$ and
\[
\Gamma^{(k)}:=\sigma_{\alpha_k}\big(\Gamma^{(k-1)}\big),\qquad k=1,\dots,M,
\]
then the following hold:
\begin{enumerate}
  \item\label{geo1} $\alpha_k\in \Delta(\Gamma^{(k-1)})$ for each $k=1,\dots,M$;
  \item\label{geo2} $\Gamma^{(M)}=\Gamma_1$.
\end{enumerate}
\end{lemma}
\begin{proof}
If $\Gamma_1=\Gamma_2$, we take $M=0$ and there is nothing to prove. Assume $\Gamma_1\neq \Gamma_2$.
Since the Weyl group $G$ acts transitively on the set of Weyl chambers (see \cite[Theorem 2.12 (2)]{HeckmanCoxeterGroups}), there exists
$g\in G$ such that $\Gamma_1=g(\Gamma_2)$.

Let $\Delta(\Gamma_2)=\{\beta_1,\beta_2,\ldots,\beta_{N_0}\}$ be the simple system associated with $\Gamma_2$.
By \cite[Theorem 2.12(4)]{HeckmanCoxeterGroups}, the Weyl group $G$ is generated by the simple reflections $\sigma_{\beta_1},\sigma_{\beta_2},\ldots,\sigma_{\beta_{N_0}}$. Hence, there exist indices
$i_1,\ldots,i_M\in\{1,\ldots,N_0\}$ such that
\[
g=\sigma_{\beta_{i_1}}\sigma_{\beta_{i_2}}\cdots \sigma_{\beta_{i_M}}.
\]
Define
\[
w_0:=\mathrm{Id},\qquad w_k:=\sigma_{\beta_{i_1}}\cdots \sigma_{\beta_{i_k}}\quad (k=1,\ldots,M),
\]
and set
\[
\Gamma^{(k)}:=w_k(\Gamma_2)\qquad (k=0,1,\ldots,M).
\]
It follows that $\Gamma^{(0)}=\Gamma_2$ and $\Gamma^{(M)}=w_M(\Gamma_2)=g(\Gamma_2)=\Gamma_1$, which implies the statement \eqref{geo2}.

For each $k=1,\ldots,M$, define
\[
\alpha_k:=w_{k-1}(\beta_{i_k})\in R.
\]
The rest of the proof is divided into two steps.

{\bf $\bullet$ Step 1.} {\it Verify that $\Gamma^{(k)}=\sigma_{\alpha_k}(\Gamma^{(k-1)})$.}

Since each $w\in G$ is orthogonal, for any root $\beta\in R$ one has the conjugation identity
\[
w\,\sigma_{\beta}\,w^{-1}=\sigma_{w(\beta)},\qquad w\in G.
\]
Applying this with $w=w_{k-1}$ and $\beta=\beta_{i_k}$ yields
\[
\sigma_{\alpha_k}=\sigma_{w_{k-1}(\beta_{i_k})}=w_{k-1}\sigma_{\beta_{i_k}}w_{k-1}^{-1}.
\]
Therefore,
\[
\sigma_{\alpha_k}(\Gamma^{(k-1)})
= w_{k-1}\sigma_{\beta_{i_k}}w_{k-1}^{-1}\big(w_{k-1}(\Gamma_2)\big)
= w_{k-1}\sigma_{\beta_{i_k}}(\Gamma_2)
= w_k(\Gamma_2)
= \Gamma^{(k)}.
\]

{\bf $\bullet$ Step 2.} {\it Verify that $\alpha_k\in\Delta(\Gamma^{(k-1)})$.}

Recall that if $\Gamma$ is a Weyl chamber associated with simple system $\Delta(\Gamma)$, then
\[
\Gamma=\{x\in\mathbb R^N:\ \langle x,\gamma\rangle\ge 0\ \text{for all }\gamma\in\Delta(\Gamma)\}.
\]
Applying $w_{k-1}\in G$ to the above description for $\Gamma_2$ and using
$\langle w_{k-1}x,\,\beta\rangle=\langle x,\,w_{k-1}^{-1}\beta\rangle$, we obtain
\[
\Gamma^{(k-1)}=w_{k-1}(\Gamma_2)
=\Bigl\{x\in\mathbb R^N:\ \langle x,\,w_{k-1}(\beta_j)\rangle\ge 0\ \text{for }j=1,\ldots,N_0\Bigr\}.
\]
By the uniqueness of the simple system associated with a Weyl chamber
(see Lemma~\ref{lem:simple_system_associated_to_chamber}), we conclude that
\[
\Delta(\Gamma^{(k-1)})=\Delta\big(w_{k-1}(\Gamma_2)\big)=w_{k-1}\big(\Delta(\Gamma_2)\big)=\{w_{k-1}(\beta_1),\ldots,w_{k-1}(\beta_{N_0})\}.
\]
In particular, $\alpha_k=w_{k-1}(\beta_{i_k})\in\Delta(\Gamma^{(k-1)})$, proving the first statement \eqref{geo1}.

This completes the proof of Lemma \ref{lem:gallery-simple}.
\end{proof}


\begin{definition}
We say that $R'\subset\Omega$ is an axis--parallel rectangle if
\[
R' := I_1\times\cdots\times I_N,
\]
with $I_i=[I_i^-,I_i^+]\subset[0,\infty)$ for $1\le i\le N_{0}$ and $I_i=[I_i^-,I_i^+]\subset\mathbb{R}$ for $i>N_{0}$.
\end{definition}

Denote by
$$B_\Omega(x_0,r):=\{x\in\Omega:\ \|x-x_0\|<r\}.$$
The following lemma demonstrates that $(\Omega,\|\cdot\|,d\mu)$ is a doubling metric measure space.
\begin{lemma}\label{lem:ball-2sided-N}
There exists a constant $C\ge 1$ such that
for every $x_0=(u_0,z_0)\in\Omega$ and every $r>0$,
\begin{equation}\label{eq:ball-2sided-Omega}
C^{-1}\,r^N\prod_{v\in R_+}\Big(L_{v,0}+r\Big)^{2k(v)}
\ \le\
\mu\big(B_\Omega(x_0,r)\big)
\ \le\
C\,r^N\prod_{v\in R_+}\Big(L_{v,0}+r\Big)^{2k(v)},
\end{equation}
where
\[
L_v(u):=\sum_{i=1}^{N_{0}} a_{v,i}u_i,
\qquad
L_{v,0}:=L_v(u_0)=\sum_{i=1}^{N_{0}} a_{v,i}(u_0)_i.
\]
In particular, there exists a constant $C_{\mathrm{dbl},\Omega}\geq 1$ such that for every $x\in\Omega$ and every $r>0$,
\[
\mu\big(B_\Omega(x,2r)\big)\ \le\ C_{\mathrm{dbl},\Omega}\;\mu\big(B_\Omega(x,r)\big).
\]
\end{lemma}

\begin{proof}
Fix $x_0=(u_0,z_0)\in\Omega$ and $r>0$. For each $v\in R_+$ define the linear form
\[
L_v(u):=a_v\cdot u,\qquad a_v:=(a_{v,1},\dots,a_{v,N_{0}})\in \mathbb{R}_+^{N_{0}}\setminus\{0\}.
\]
Since $R_+$ is finite and each $a_v\neq 0$, we have
\[
L_*:=\max_{v\in R_+}\|a_v\|<\infty,
\qquad
m_*:=\min_{v\in R_+}\sum_{i=1}^{N_{0}} a_{v,i}>0.
\]

\noindent
\textbf{Upper bound.}
Let $x=(u,z)\in B_\Omega(x_0,r)$. Then
\[
\|u-u_0\|\ \le\ \|x-x_0\|\ \le\ r.
\]
Hence, for every $v\in R_+$,
\[
|L_v(u)-L_{v,0}|
=\big|a_v\cdot (u-u_0)\big|
\le \|a_v\|\cdot\|u-u_0\|
\le L_*\,r.
\]
Since $L_v(u)\ge 0$ on $\Omega$ (all coefficients and coordinates are nonnegative), we obtain
\[
L_v(u)\le L_{v,0}+L_*r\le C_1\,(L_{v,0}+r),
\qquad
C_1:=\max\{1,L_*\}.
\]
Therefore,
\[
\phi(u)
=\prod_{v\in R_+} L_v(u)^{2k(v)}
\le C_1^{2\sum_{v\in R_+}k(v)}\prod_{v\in R_+}(L_{v,0}+r)^{2k(v)}
=:C_2\prod_{v\in R_+}(L_{v,0}+r)^{2k(v)}.
\]
Integrating over $B_\Omega(x_0,r)$ and using
$|B_\Omega(x_0,r)|\le |B(x_0,r)|=\omega_N r^N$, where $\omega_N$ is the volume of the unit ball in $\mathbb{R}^N$, gives
\[
\mu\big(B_\Omega(x_0,r)\big)
=\int_{B_\Omega(x_0,r)}\phi(u)\,dx
\le C_2\,|B(x_0,r)|\,\prod_{v\in R_+}(L_{v,0}+r)^{2k(v)}
\le C_3\,r^N\prod_{v\in R_+}(L_{v,0}+r)^{2k(v)},
\]
with $C_3:=C_2\,\omega_N$.

\medskip\noindent
\textbf{Lower bound.}
We construct a smaller Euclidean ball $B_c$ contained in $B_\Omega(x_0,r)$ on which $\phi$ admits a uniform
positive lower bound in terms of $\prod_{v\in R_+}(L_{v,0}+r)^{2k(v)}$. To this end, we first choose
\[
\lambda:=1+\frac{L_*}{m_*}>1,
\qquad
\eta:=\frac{1}{4\big(\sqrt{N_{0}}\,\lambda+1\big)},
\qquad
\theta:=\lambda\,\eta.
\]
These parameters depend only on $L_*$ and $m_*$, and satisfy
\[
\theta> \eta>0,
\qquad
\sqrt{N_{0}}\,\theta+\eta=\eta(\sqrt{N_{0}}\lambda+1)=\frac14<1.
\]
Let $\mathbf{1}_{N_{0}}:=(1,\dots,1)\in\mathbb{R}^{N_{0}}$ and define the center
\[
x_c:=(u_c,z_c):=\big(u_0+\theta r\,\mathbf{1}_{N_{0}},\ z_0\big),
\qquad
B_c:=B(x_c,\eta r).
\]
We {\it claim} that $B_c\subset B_\Omega(x_0,r)$. To see this inclusion, we first apply the triangle inequality to see that for any $x\in B_c$,
\[
\|x-x_0\|\le \|x-x_c\|+\|x_c-x_0\|
\le \eta r+\theta r\sqrt{N_{0}}
\le r,
\]
since $\sqrt{N_{0}}\theta+\eta<1$. Hence $B_c\subset B(x_0,r)$. Moreover, for any $x=(u,z)\in B_c$, we have $\|u-u_c\|\le \|x-x_c\|\le \eta r$. This implies that for each $1\le i\le N_{0}$,
\[
u_i\ge (u_c)_i-\eta r=(u_0)_i+(\theta-\eta)r\ge 0,
\]
since $u_0\in\mathbb{R}_+^{N_{0}}$ and $\theta\ge\eta$. Thus $u\in\mathbb{R}_+^{N_{0}}$ and $x\in\Omega$. Therefore,  $B_c\subset B_\Omega(x_0,r)$. This finishes  the proof of the  claim.

\smallskip
Next we obtain a uniform lower bound for $L_v(u)$ on $B_c$.
Write any $x=(u,z)\in B_c$ as
\[
u=u_0+\theta r\,\mathbf{1}_{N_{0}}+\Delta_u,
\qquad
z=z_0+\Delta_z,
\qquad
\|\Delta_u\|^2+\|\Delta_z\|^2\le (\eta r)^2,
\]
so in particular $\|\Delta_u\|\le \eta r$.
Then for all $x=(u,z)\in B_c$ and all $v\in R_+$,
\begin{align*}
L_v(u)
&=a_v\cdot u
=a_v\cdot u_0+\theta r\,a_v\cdot\mathbf{1}_{N_{0}}+a_v\cdot\Delta_u \\
&\ge L_{v,0}+\theta r\sum_{i=1}^{N_{0}} a_{v,i}-\|a_v\|\,\|\Delta_u\|
\ge L_{v,0}+rK_0\geq c_0(L_{v,0}+r).
\end{align*}
with
\[
K_0:=\theta m_*-\eta L_*=\eta(\lambda m_*-L_*)=\eta m_*>0,\qquad c_0:=\min\{1,K_0\}\in(0,1].
\]
Consequently, for all $u\in B_c$,
\[
\phi(u)
=\prod_{v\in R_+}L_v(u)^{2k(v)}
\ge c_0^{2\sum_{v\in R_+}k(v)}\prod_{v\in R_+}(L_{v,0}+r)^{2k(v)}
=:C_4\prod_{v\in R_+}(L_{v,0}+r)^{2k(v)}.
\]
This, in combination with the fact that $B_c\subset B_\Omega(x_0,r)$, yields
\[
\mu\big(B_\Omega(x_0,r)\big)
\ge \mu(B_c)
=\int_{B_c}\phi(u)\,dx
\ge C_4\,|B_c|\prod_{v\in R_+}(L_{v,0}+r)^{2k(v)}
= C_5\,r^N\prod_{v\in R_+}(L_{v,0}+r)^{2k(v)},
\]
where $C_5:=C_4\,\omega_N\eta^N>0$.

The second statement follows immediately from the first one. Indeed, by the first statement,
\[
\mu\big(B_\Omega(x_0,2r)\big)
\le C_3\,(2r)^N\prod_{v\in R_+}(L_{v,0}+2r)^{2k(v)}
\le C_3\,2^{N+\mathcal{K}}\,r^N\prod_{v\in R_+}(L_{v,0}+r)^{2k(v)}
\le C_3C_5^{-1}\,2^{N+\mathcal{K}}\,\mu\big(B_\Omega(x_0,r)\big),
\]
where we set $\mathcal{K}:=2\sum_{v\in R_+}k(v)$. Therefore, $(\Omega,\|\cdot\|,\mu)$ is doubling with the doubling constant $C_{\mathrm{dbl},\Omega}:=C_3C_5^{-1}\,2^{N+\mathcal{K}}$.
 This finishes the proof of Lemma \ref{lem:ball-2sided-N}.
\end{proof}
Let $\Phi:\mathbb{R}^N\longrightarrow\mathbb{R}^N$ be the linear isomorphism given in  Corollary~\ref{coro:chamber-orthant} such that
$
\Phi(\Gamma)=\Omega.
$
Since $\Phi$ is linear and invertible, there exist positive constants $c_\Phi$ and $C_\Phi$ such that for any $x,y\in\mathbb{R}^N$,
\begin{equation}\label{eq:biLip-N}
c_\Phi\,\|x-y\|\ \le\ \|\Phi(x)-\Phi(y)\|\ \le\ C_\Phi\,\|x-y\|.
\end{equation}
Denote by $B_\Gamma(x,r):=\{y\in\Gamma:\ \|y-x\|<r\}$. It follows from \eqref{eq:biLip-N} that for any $x_0\in\Gamma$ and $r>0$,
\begin{align}\label{beforedoublingball}
B_{\Omega}(\Phi(x_0),c_\Phi r)\subset \Phi(B_{\Gamma}(x_0,r))\subset B_{\Omega}(\Phi(x_0),C_\Phi r).
\end{align}
This, together with Lemma  \ref{lem:ball-2sided-N}, yields
\begin{align}\label{doublingball}
  \mu(\Phi(B_{\Gamma}(x_0,r))) \simeq \mu(B_{\Omega}(\Phi(x_0),C_\Phi r)).
\end{align}
\begin{lemma}\label{construction-rectangle}
  For all $x_0\in\Gamma$ and $r>0$, one can find an axis--parallel rectangle $R'$ on $\Omega$ such that
\begin{align}\label{step3goal}
B_{\Omega}(\Phi(x_0),C_\Phi r)\subset R'\subset B_{\Omega}(\Phi(x_0),\sqrt{N}C_\Phi r).
\end{align}
Furthermore, we have
\begin{align}\label{Rdoubling}
  \mu(R')\simeq \mu(B_{\Omega}(\Phi(x_0),C_\Phi r)).
\end{align}
%\begin{align}\label{step3goal}
%\Phi(B_\Gamma(x_0,r))\subset R'\quad \text{and}\quad
%\mu(\Phi(B_\Gamma(x_0,r)))\simeq \mu(R').
%\end{align}
\end{lemma}
\begin{proof}
We write
$
(u_0,z_0):=\Phi(x_0)\in\Omega.
$
and define the axis--parallel rectangle
\begin{equation}\label{eq:rectangle-choice}
R':=\Big(\prod_{i=1}^{N_{0}} \big[\max\{0,(u_0)_i-C_\Phi r\},\ (u_0)_i+C_\Phi r\big]\Big)
\times
\Big(\prod_{j=1}^{N-N_{0}}\big[(z_0)_j-C_\Phi r,\ (z_0)_j+C_\Phi r\big]\Big).
\end{equation}

On the one hand, it follows immediately that $B_\Omega(\Phi(x_0),C_\Phi r)\subset R'$. On the other hand, since $R'$ is a rectangle with half-side lengths bounded by $C_\Phi r$ in each coordinate, we have
\begin{align*}
R'\subset B_\Omega(\Phi(x_0),\sqrt{N}\,C_\Phi r).
\end{align*}
The second statement follows from the first one together with Lemma \ref{lem:ball-2sided-N}. The proof of Lemma \ref{construction-rectangle} is finished.
\end{proof}
The following theorem demonstrates that $(\Gamma,\|\cdot\|,dw)$ is a doubling metric measure space.


\begin{theorem}\label{main:dunkldoubling}
There exists a constant $C\ge 1$ such that
for every $x\in\Gamma$ and every $r>0$,
\begin{equation}\label{eq:ball-2sided-Gamma}
C^{-1}\,r^N \prod_{v \in R_+}(|\langle x, v \rangle |+r)^{2k(v)}
\ \le\
w\big(B_\Gamma(x,r)\big)
\ \le\
C\,r^N \prod_{v \in R_+}(|\langle x, v \rangle |+r)^{2k(v)}.
\end{equation}
In particular, there exists a constant $C_{\mathrm{dbl},\Gamma}\geq 1$ such that for every $x\in\Gamma$ and every $r>0$,
\[
w\big(B_\Gamma(x,2r)\big)\ \le\ C_{\mathrm{dbl},\Gamma}\;w\big(B_\Gamma(x,r)\big).
\]
Moreover, for all $x\in\Gamma$ and $r>0$, we have
\begin{align}\label{globallocalvol}
w(B_\Gamma(x,r))\sim w(B(x,r)).
\end{align}
\end{theorem}
\begin{proof}
The main strategy of the proof is to transfer the ball estimates from $(\Omega,\|\cdot\|,\mu)$ back to $(\Gamma,\|\cdot\|,dw)$. Fix $x\in\Gamma$ and $r>0$. It follows from the equality \eqref{defmu} that
\begin{equation}\label{eq:pushforward-ball}
w\big(B_\Gamma(x,r)\big)=\mu\big(\Phi(B_\Gamma(x,r))\big).
\end{equation}
Applying $\mu$ to the inclusions \eqref{beforedoublingball} and using \eqref{eq:pushforward-ball} yields
\begin{equation}\label{eq:wball-between-muballs}
\mu\big(B_{\Omega}(\Phi(x),c_\Phi r)\big)
\ \le\
w\big(B_\Gamma(x,r)\big)
\ \le\
\mu\big(B_{\Omega}(\Phi(x),C_\Phi r)\big).
\end{equation}
Write $\Phi(x)=(u,z)\in\Omega$. For each $v\in R_+$ recall the linear form
$L_{v,0}:=\sum_{i=1}^{N_0}a_{v,i}u_i$.
Combining the inequality \eqref{eq:wball-between-muballs} with Lemma~\ref{lem:ball-2sided-N} and the equality \eqref{changevariable}, we obtain that
\begin{equation}\label{eq:wball-in-terms-of-L}
w\big(B_\Gamma(x,r)\big)\ \simeq\ r^N\prod_{v\in R_+}\big(L_{v,0}+r\big)^{2k(v)}=r^N\prod_{v\in R_+}\big(|\langle x,v\rangle|+r\big)^{2k(v)}.
\end{equation}
This proves \eqref{eq:ball-2sided-Gamma}. The second statement follows immediately from \eqref{eq:ball-2sided-Gamma}. The third statement follows from the inequalities \eqref{eq:ball-2sided-Gamma} and \eqref{twosideinequ}. This completes the proof of Theorem~\ref{main:dunkldoubling}.
\end{proof}

$
Let
\[
\mathbb D_\Gamma
=
\bigcup_{j\in\mathbb Z}\mathbb D_{\Gamma,j}
\]
be the dyadic system over $\Gamma$ associated with this choice of $\Phi$
as in Corollary~\ref{Construction over gamma}. {\color{red}[This paragraph of `recall' is just for Dima's convenience, which can be finally erased]}

For each $I\in\mathbb D_{\Gamma,j}$, let
$Q\in\mathbb D^0_{\Omega,j}$ be the unique dyadic cube such that
$
I=\Phi^{-1}(Q).
$
Then
$
\ell(I)=\ell(Q)=2^{-j}.
$
We define
\[
I^\ast:=\Phi^{-1}\bigl((2Q)\cap\Omega\bigr),
\qquad
z_I:=\Phi^{-1}(c_Q).
\]
\end{nota}
\begin{lemma}\label{lem:Dunkl-Riesz-endpoint-local-geometry}
Fix $\sigma,\tau\in G$ and $I=\Phi^{-1}(Q)\in\mathbb D_\Gamma$. There exists $\varepsilon_0>0$ independent of $\sigma,\tau$ and $I$, such that, whenever
\[
\operatorname{dist}\left(
(a,b),
\left\{
(a_0,b_0):
a_0\in[-1/2,1/2]^N,\
c_Q+\ell(I) b_0\in(2Q)\cap\Omega
\right\}
\right)<\varepsilon_0
\]
and
\[
\frac18\leq |a-b|\leq\frac12,
\]
the points
$
X=\sigma\Phi^{-1}(c_Q+\ell(I) a)$ and
$
Y=\tau\Phi^{-1}(c_Q+\ell(I) b)
$
satisfy
\[
d(X,Y)\simeq\ell(I),
\qquad
\|X-Y\|\simeq\ell(I)+|\sigma z_I-\tau z_I|,
\qquad
V(X,Y,d(X,Y))\simeq w(I).
\]
\end{lemma}

\begin{proof}
Choose $a_0$ and $b_0$ such that $a_0\in[-1/2,1/2]^N$, $
c_Q+\ell(I) b_0\in(2Q)\cap\Omega$, and
$|a-a_0|+|b-b_0|\lesssim\varepsilon_0$ for some $\varepsilon_0$ to be chosen later. Set $
X_0=\sigma\Phi^{-1}(c_Q+\ell(I) a_0)$ and $
Y_0=\tau\Phi^{-1}(c_Q+\ell(I) b_0).$
If $\varepsilon_0$ is sufficiently small, then
$|a_0-b_0|\gtrsim1$. Since
$c_Q+\ell(I) a_0,c_Q+\ell(I) b_0\in\Omega$, we apply the $G$-invariance of $d$,
\eqref{distanceequa}, and \eqref{eq:biLip-N} to see that
\[
d(X_0,Y_0)
=
\left\|
\Phi^{-1}(c_Q+\ell(I) a_0)
-
\Phi^{-1}(c_Q+\ell(I) b_0)
\right\|
\gtrsim\ell(I).
\]
Since $\Phi^{-1}$ is Lipschitz,
\[
d(X,Y)
\geq
d(X_0,Y_0)-\|X-X_0\|-\|Y-Y_0\|
\gtrsim \ell(I)-C\varepsilon_0\ell(I)
\gtrsim\ell(I)
\]
after decreasing $\varepsilon_0$ if necessary. Conversely, by the $G$-invariance of $d$ and \eqref{eq:biLip-N},
\[
d(X,Y)
\leq
\left\|
\Phi^{-1}(c_Q+\ell(I) a)
-
\Phi^{-1}(c_Q+\ell(I) b)
\right\|
\lesssim
\ell(I)|a-b|
\lesssim\ell(I).
\]
Thus, we have $d(X,Y)\simeq\ell(I)$.

The support condition also gives $|a|+|b|\lesssim1$, and hence,
\[
\|X-\sigma z_I\|+\|Y-\tau z_I\|=\|\Phi^{-1}(c_Q+\ell(I) a)-\Phi^{-1}(c_Q)\|+\|\Phi^{-1}(c_Q+\ell(I) b)-\Phi^{-1}(c_Q)\|\lesssim\ell(I).
\]
Therefore
\begin{align}\label{cbin111}
\|X-Y\|
\lesssim
\ell(I)+|\sigma z_I-\tau z_I|.
\end{align}
On the other hand,
\[
\ell(I)\lesssim d(X,Y)\leq\|X-Y\|
\]
and
\[
|\sigma z_I-\tau z_I|
\leq
\|X-Y\|+\|X-\sigma z_I\|+\|Y-\tau z_I\|
\lesssim
\|X-Y\|.
\]
Hence,
\[
\|X-Y\|\simeq\ell(I)+|\sigma z_I-\tau z_I|,
\]

By \eqref{eq:biLip-N}, there are fixed constants $c,C>0$ such that
\[
B_\Gamma(z_I,c\ell(I))\subset I\subset B_\Gamma(z_I,C\ell(I)).
\]
Thus \eqref{doublinginequality} and \eqref{globallocalvol} imply
\[
w(B(z_I,\ell(I)))\simeq w(I).
\]
Since $d(X,Y)\simeq\ell(I)$ and
$\|X-\sigma z_I\|+\|Y-\tau z_I\|\lesssim\ell(I)$, we apply the doubling property and the $G$-invariance of $dw$ to see that
\[
w(B(X,d(X,Y)))
\simeq
w(B(\sigma z_I,\ell(I)))
=
w(B(z_I,\ell(I)))
\simeq
w(I),
\]
and similarly,
\[
w(B(Y,d(X,Y)))\simeq w(I).
\]
It follows that
\[
V(X,Y,d(X,Y))\simeq w(I).
\]
This completes the proof.
\end{proof}

%{\color{red}The following Lemma demonstrates that Dunkl--Riesz transform kernel can be decomposed locally into a summation of NWO sequences with a separate variable form and a suitable decay factor.}
\begin{lemma}
\label{lem:Dunkl-Riesz-Fourier-NWO-endpoint}
Fix a closed Weyl chamber \(\Gamma\). Fix \(j\in\{1,\ldots,N\}\).
For \(\sigma,\tau\in G\) and $I\in\mathbb D_{\Gamma}$, define
\begin{equation}
\label{eq:D-and-theta}
 \theta_{\sigma,\tau,I}
 =
 \frac{\ell(I)}{\ell(I)+|\sigma z_I-\tau z_I|}.
\end{equation}
Then there exist a sequence of real numbers $\{c_{u,v}\}_{u,v\in\mathbb Z^N}$ and two families of functions $$\{e_{u,v,I}^{\sigma,\tau}\}_{u,v\in\mathbb Z^N,\,\sigma,\tau\in G,\,I\in\mathbb D_{\Gamma}}\qquad {\rm and}\qquad \{h_{u,v,I}^{\sigma,\tau}\}_{u,v\in\mathbb Z^N,\,\sigma,\tau\in G,\,I\in\mathbb D_{\Gamma}}$$ such that
\begin{enumerate}
  \item\label{property1} $
|c_{u,v}|
\lesssim_{M_0}
(1+|u|+|v|)^{-M_0}$, for $M_0>2N;$
  \item\label{property2} $
\operatorname{supp}e_{u,v,I}^{\sigma,\tau}
\subseteq\sigma I,\
\operatorname{supp}h_{u,v,I}^{\sigma,\tau}
\subseteq\tau I^*;$
  \item\label{property3} $
|e_{u,v,I}^{\sigma,\tau}|
+
|h_{u,v,I}^{\sigma,\tau}|
\lesssim
w(I)^{-1/2};
$
  \item for almost every \((x,y)\) with \(d(x,y)>0\),
\begin{equation}
\label{eq:endpoint-Dunkl-Riesz-kernel-NWO}
K_{{\rm Dunkl},j}(x,y)
=
\sum_{\sigma,\tau\in G}
\sum_{u,v\in\mathbb Z^N}
c_{u,v}
\sum_{I\in\mathbb D_\Gamma}
\theta_{\sigma,\tau,I}
e_{u,v,I}^{\sigma,\tau}(x)
h_{u,v,I}^{\sigma,\tau}(y).
\end{equation}
\end{enumerate}
\end{lemma}
\begin{proof}
Fix $\sigma,\tau\in G$. For
$x\in\sigma(\Gamma)^\circ$ and $y\in\tau(\Gamma)^\circ$, write
$
\xi=\Phi(\sigma^{-1}x),
$
and
$
\eta=\Phi(\tau^{-1}y).
$
By the $G$-invariance of the orbit distance, \eqref{distanceequa}, and
\eqref{eq:biLip-N},
\[
d(x,y)
=
\|\sigma^{-1}x-\tau^{-1}y\|
\simeq
|\xi-\eta|.
\]

Choose $\psi\in C_c^\infty(\mathbb R^N)$ such that
\begin{align}\label{supportcondi}
\operatorname{supp}\psi
\subset
\left\{z:\frac18\leq |z|\leq\frac12\right\},
\qquad
\sum_{m\in\mathbb Z}\psi(2^mz)=1,
\quad z\neq0.
\end{align}
If \(d(x,y)>0\), then for each \(m\in\mathbb Z\), there is a unique
\(I=\Phi^{-1}(Q)\in\mathbb D_{\Gamma,m}\) with
\(Q\in\mathbb D_{\Omega,m}^{0}\) such that
\(\sigma^{-1}x\in I\). It therefore follows from
\eqref{supportcondi} that
\begin{equation*}
K_{{\rm Dunkl},j}(x,y)
=
\sum_{I\in\mathbb D_\Gamma}
\chi_{\sigma I}(x)
K_{{\rm Dunkl},j}(x,y)
\psi\left(
\frac{\Phi(\sigma^{-1}x)-\Phi(\tau^{-1}y)}
{\ell(I)}
\right).
\end{equation*}
Moreover, if
$x\in\sigma I$ and
\[
\psi\left(
\frac{\Phi(\sigma^{-1}x)-\Phi(\tau^{-1}y)}
{\ell(I)}
\right)\neq0,
\]
then $|\xi-\eta|\leq\ell(I)/2$. Since $\xi\in Q$ and
$\eta\in\Omega$, this implies $\eta\in(2Q)\cap\Omega$, and hence
\begin{align}\label{yistar}
y\in\tau I^*.
\end{align}

Fix $I=\Phi^{-1}(Q)$ and write $\ell=\ell(I)$. By the construction of
$\mathbb D_\Omega^0$, we may write
\[
Q=\prod_{i=1}^N[m_i\ell,(m_i+1)\ell),
\]
where $m_i\in\mathbb Z_+$ for $1\leq i\leq N_0$ and
$m_i\in\mathbb Z$ for $N_0<i\leq N$. Since the $i$-th coordinate of
$c_Q$ is $(m_i+\frac12)\ell$, the condition
$c_Q+\ell b\in(2Q)\cap\Omega$ gives
$
-1\leq b_i<1
$
for every $i$, with the additional restriction
$-\frac12\leq b_i<1$ precisely when $1\leq i\leq N_0$ and $m_i=0$.
Consequently, the sets
\[
\left\{
(a,b):
a\in[-1/2,1/2]^N,\
c_Q+\ell b\in(2Q)\cap\Omega
\right\}
\]
range over at most $2^{N_0}$ rectangles. Let $\varepsilon_0$ be as in
Lemma~\ref{lem:Dunkl-Riesz-endpoint-local-geometry}, decreased if
necessary so that the $\varepsilon_0$-neighborhood of each of these
rectangles is contained in $(-2,2)^N\times(-2,2)^N$. Since there are
only finitely many such rectangles, we may choose
$\chi_Q\in C_c^\infty((-2,2)^N\times(-2,2)^N)$, equal to $1$ on the
corresponding rectangle and supported in its $\varepsilon_0$-neighborhood,
so that
\[
\sup_Q\|\chi_Q\|_{C^L(\mathbb R^{2N})}<\infty
\]
for every $L\geq0$.

For
$
X=\sigma\Phi^{-1}(c_Q+\ell a),
$
and
$
Y=\tau\Phi^{-1}(c_Q+\ell b),
$
it follows from Lemmas~\ref{lem:Dunkl-Riesz-endpoint-local-geometry} and
\ref{lem:Dunkl-Riesz-endpoint-derivatives} that for every pair of multi-indices $\alpha,\beta$,
\[
\left|
\partial_x^\alpha\partial_y^\beta
K_{{\rm Dunkl},j}(X,Y)
\right|
\lesssim
\theta_{\sigma,\tau,I}
\frac{\ell^{-|\alpha|-|\beta|}}{w(I)}.
\]

Define
\[
H_{\sigma,\tau,I}(a,b)
=
\frac{w(I)}{\theta_{\sigma,\tau,I}}
\chi_Q(a,b)\psi(a-b)
K_{{\rm Dunkl},j}
\left(
\sigma\Phi^{-1}(c_Q+\ell a),
\tau\Phi^{-1}(c_Q+\ell b)
\right).
\]
The preceding derivative estimate, the chain rule, and the uniform
bounds for $\chi_Q$ imply
\[
\sup_{\sigma,\tau\in G}
\sup_{I\in\mathbb D_\Gamma}
\|H_{\sigma,\tau,I}\|_{C^L(\mathbb R^{2N})}
<\infty,
\qquad L\geq0.
\]
Since $H_{\sigma,\tau,I}$ is compactly supported in
$(-2,2)^N\times(-2,2)^N$, its $4$-periodic extension has the Fourier
expansion
\[
H_{\sigma,\tau,I}(a,b)
=
\sum_{u,v\in\mathbb Z^N}
b_{u,v,I}^{\sigma,\tau}
e^{\frac{\pi i}{2}u\cdot a}
e^{\frac{\pi i}{2}v\cdot b},
\]
and repeated integration by parts yields, for every $L\geq1$,
\[
\sup_{\sigma,\tau\in G}
\sup_{I\in\mathbb D_\Gamma}
|b_{u,v,I}^{\sigma,\tau}|
\lesssim_L
(1+|u|+|v|)^{-L}.
\]
Choosing an integer $L>M_0$, there is a constant $C_{M_0}$ such that
\[
|b_{u,v,I}^{\sigma,\tau}|
\leq
C_{M_0}(1+|u|+|v|)^{-M_0}=:c_{u,v}
\]
uniformly in $\sigma,\tau$ and $I$.
For $I=\Phi^{-1}(Q)$, define
\begin{align*}
e_{u,v,I}^{\sigma,\tau}(x)
&=
\frac{b_{u,v,I}^{\sigma,\tau}}{c_{u,v}}
w(I)^{-1/2}\chi_{\sigma I}(x)
\exp\left(
\frac{\pi i}{2\ell(I)}
u\cdot\bigl(\Phi(\sigma^{-1}x)-c_Q\bigr)
\right),\\
h_{u,v,I}^{\sigma,\tau}(y)
&=
w(I)^{-1/2}\chi_{\tau I^*}(y)
\exp\left(
\frac{\pi i}{2\ell(I)}
v\cdot\bigl(\Phi(\tau^{-1}y)-c_Q\bigr)
\right).
\end{align*}
Then \eqref{property2} holds, and
\[
|e_{u,v,I}^{\sigma,\tau}|
\leq
w(I)^{-1/2},\qquad |h_{u,v,I}^{\sigma,\tau}|\leq w(I)^{-1/2},
\]
which gives \eqref{property3}.

If $x\in\sigma I$ and $y\in\tau I^*$, set
\[
a=\frac{\Phi(\sigma^{-1}x)-c_Q}{\ell(I)},
\qquad
b=\frac{\Phi(\tau^{-1}y)-c_Q}{\ell(I)}.
\]
Since $\sigma^{-1}x\in I=\Phi^{-1}(Q)$,
$a\in[-1/2,1/2]^N$, while
\[
c_Q+\ell(I)b
=
\Phi(\tau^{-1}y)
\in(2Q)\cap\Omega
\]
because $\tau^{-1}y\in I^*$. Hence $\chi_Q(a,b)=1$. The Fourier expansion
therefore gives
\begin{align*}
&K_{{\rm Dunkl},j}(x,y)
\psi\left(
\frac{\Phi(\sigma^{-1}x)-\Phi(\tau^{-1}y)}
{\ell(I)}
\right)=
\sum_{u,v\in\mathbb Z^N}
c_{u,v}\theta_{\sigma,\tau,I}
e_{u,v,I}^{\sigma,\tau}(x)
h_{u,v,I}^{\sigma,\tau}(y).
\end{align*}
If $x\in\sigma I$ and $y\notin\tau I^*$, then
$h_{u,v,I}^{\sigma,\tau}(y)=0$. On the other hand, if
\[
\psi\left(
\frac{\Phi(\sigma^{-1}x)-\Phi(\tau^{-1}y)}
{\ell(I)}
\right)\neq0,
\]
then \eqref{supportcondi} gives $|\xi-\eta|\leq\ell(I)/2$, and hence,
as above, $\eta\in(2Q)\cap\Omega$, which would imply
$y\in\tau I^*$ by \eqref{yistar}. Thus the $\psi$-factor vanishes as
well. Hence, for almost every
$(x,y)\in\sigma\Gamma\times\tau\Gamma$,
\begin{align*}
&\chi_{\sigma I}(x)K_{{\rm Dunkl},j}(x,y)
\psi\left(
\frac{\Phi(\sigma^{-1}x)-\Phi(\tau^{-1}y)}
{\ell(I)}
\right)=
\sum_{u,v\in\mathbb Z^N}
c_{u,v}\theta_{\sigma,\tau,I}
e_{u,v,I}^{\sigma,\tau}(x)
h_{u,v,I}^{\sigma,\tau}(y).
\end{align*}
Summing this localized identity over
$I\in\mathbb D_\Gamma$ and using the dyadic partition of unity, we obtain
\[
K_{{\rm Dunkl},j}(x,y)
=
\sum_{u,v\in\mathbb Z^N}
c_{u,v}
\sum_{I\in\mathbb D_\Gamma}
\theta_{\sigma,\tau,I}
e_{u,v,I}^{\sigma,\tau}(x)
h_{u,v,I}^{\sigma,\tau}(y)
\]
for almost every
$x\in\sigma(\Gamma)^\circ$, $y\in\tau(\Gamma)^\circ$ with
$d(x,y)>0$.

The interiors of the Weyl chambers cover $\mathbb R^N$ up to the
reflecting hyperplanes, which have $dw$-measure zero. Since
\[
\operatorname{supp}e_{u,v,I}^{\sigma,\tau}\subset\sigma\Gamma,
\qquad
\operatorname{supp}h_{u,v,I}^{\sigma,\tau}\subset\tau\Gamma,
\]
and the interiors of distinct Weyl chambers are disjoint, for almost every
$(x,y)\in\mathbb R^N\times\mathbb R^N$ only the unique pair
$(\sigma,\tau)$ satisfying
$x\in\sigma(\Gamma)^\circ$ and $y\in\tau(\Gamma)^\circ$ contributes.
Thus, summing over $\sigma,\tau\in G$, we obtain
\eqref{eq:endpoint-Dunkl-Riesz-kernel-NWO}.
\end{proof}
\begin{lemma}
\label{lem:Dunkl-mixed-cube-to-Lebesgue-oscillation}
Fix \(s\geq1\). There exists an exponent \(r_s>s\), depending only
on \(s\) and the structural data of the Dunkl weight, such that
for every \(I\in\mathbb D_\Gamma\) and every
\(g\in L^1_{\mathrm{loc}}(\mathbb R^N)\),
\begin{equation}
\label{eq:Dunkl-mixed-cube-to-Lebesgue-oscillation}
\left(
\frac1{w(I)}
\int_I
\left|
g-(g)_{I,dx}
\right|^s\,dw
\right)^{1/s}
\lesssim
\operatorname{osc}_{r_s}(g;I).
\end{equation}
Moreover, for every \(x\in\Gamma\) and \(r>0\),
\begin{equation}
\label{Omegacontrol}
\left(
\frac{1}{w\bigl(B_\Gamma(x,r)\bigr)}
\int_{B_\Gamma(x,r)}
\left|
g-(g)_{B_\Gamma(x,r),dx}
\right|^s\,dw
\right)^{1/s}
\lesssim
\operatorname{osc}_{r_s}\bigl(g;B_\Gamma(x,r)\bigr).
\end{equation}
The implicit constants are independent of \(I\), \(g\), \(x\), and \(r\).
\end{lemma}
\begin{proof}
By Lemma~\ref{lem:Dunkl-weight-A-infty} and the reverse H\"older inequality of $A_\infty$ weights \cite[Theorem~7.3.3(c)]{MR3243734}, there exists $t>1$ such that
\[
\left(\fint_B\varpi^t\,dx\right)^{1/t}
\lesssim
\fint_B\varpi\,dx
\]
for every Euclidean ball $B$.

Fix $I\in\mathbb D_\Gamma$. By the construction of $\mathbb D_\Gamma$ and \eqref{eq:biLip-N}, there exist constants $c,C>0$ such that
\[
B_\Gamma(z_I,c\ell(I))
\subset I
\subset B_\Gamma(z_I,C\ell(I)).
\]
For $E=I$, set
$
B=B(z_I,C\ell(I)).
$
Then
\[
E\subset B,
\qquad
|E|\simeq |B|,
\qquad
w(E)\simeq w(B),
\]
where the last equivalence follows from Theorem~\ref{main:dunkldoubling} and \eqref{globallocalvol}.

For $x\in\Gamma$ and $r>0$, set
$
E=B_\Gamma(x,r)$ and $
B=B(x,r).
$
Then
\[
E\subset B,
\qquad
|E|\simeq |B|,
\qquad
w(E)\simeq w(B).
\]

Consequently, for either of the above choices of $E$ and $B$,
\[
\left(\frac1{|E|}\int_E\varpi^t\,dx\right)^{1/t}
\lesssim
\left(\frac1{|B|}\int_B\varpi^t\,dx\right)^{1/t}
\lesssim
\frac{w(B)}{|B|}
\simeq
\frac{w(E)}{|E|}.
\]

Let $r_s=st'$. Then $r_s>s$. For either choice of $E$, we apply H\"older's inequality to deduce that
\begin{align*}
\left[
\frac1{w(E)}
\int_E|g-(g)_{E,dx}|^s\,dw
\right]^{1/s}
&\leq
w(E)^{-1/s}
\left(
\int_E|g-(g)_{E,dx}|^{r_s}\,dx
\right)^{1/r_s}
\left(
\int_E\varpi^t\,dx
\right)^{1/(st)}\\
&\lesssim
\left[
\frac1{|E|}
\int_E|g-(g)_{E,dx}|^{r_s}\,dx
\right]^{1/r_s}.
\end{align*}
Taking $E=I$ and $E=B_\Gamma(x,r)$, respectively, we obtain
\eqref{eq:Dunkl-mixed-cube-to-Lebesgue-oscillation} and \eqref{Omegacontrol}.
This completes the proof of Lemma~\ref{lem:Dunkl-mixed-cube-to-Lebesgue-oscillation}.
\end{proof}

\begin{lemma}
\label{lem:Lorentz-finite-multiplicity}
Let \(0<p<\infty\), \(0<q\leq\infty\). Let \(\mathcal A\) and
\(\mathcal B\) be countable sets, and let
$
\pi:\mathcal A\longrightarrow\mathcal B
$
be a map such that
\[
\sup_{b\in\mathcal B}
\#\pi^{-1}(\{b\})
\leq M
\]
for some $M\geq 1$.
Then for every family of nonnegative numbers
\(\{\lambda_b\}_{b\in\mathcal B}\),
\begin{equation}
\label{eq:Lorentz-finite-multiplicity}
\left\|
\{\lambda_{\pi(a)}\}_{a\in\mathcal A}
\right\|_{\ell^{p,q}(\mathcal A)}
\lesssim
M^{1/p}
\left\|
\{\lambda_b\}_{b\in\mathcal B}
\right\|_{\ell^{p,q}(\mathcal B)}.
\end{equation}
The implicit constant depends only on $p$ and $q$.
\end{lemma}

\begin{proof}
For \(t>0\), write
\[
n_{\mathcal A}(t)
:=
\#\{a\in\mathcal A:\lambda_{\pi(a)}>t\},
\qquad
n_{\mathcal B}(t)
:=
\#\{b\in\mathcal B:\lambda_b>t\}.
\]
By the multiplicity assumption,
\[
n_{\mathcal A}(t)
=
\sum_{\substack{b\in\mathcal B\\ \lambda_b>t}}
\#\pi^{-1}(\{b\})
\leq
M n_{\mathcal B}(t).
\]
If \(0<q<\infty\), the distribution-function formula for Lorentz
sequence spaces gives
\begin{align*}
\left\|
\{\lambda_{\pi(a)}\}_{a\in\mathcal A}
\right\|_{\ell^{p,q}}^q
&\simeq
\int_0^\infty
\left[
t\,n_{\mathcal A}(t)^{1/p}
\right]^q
\frac{dt}{t}
\\
&\leq
M^{q/p}
\int_0^\infty
\left[
t\,n_{\mathcal B}(t)^{1/p}
\right]^q
\frac{dt}{t}
\\
&\lesssim
M^{q/p}
\left\|
\{\lambda_b\}_{b\in\mathcal B}
\right\|_{\ell^{p,q}}^q.
\end{align*}
Taking the \(q\)-th root proves
\eqref{eq:Lorentz-finite-multiplicity}. If \(q=\infty\), then
\[
\left\|
\{\lambda_{\pi(a)}\}
\right\|_{\ell^{p,\infty}}
=
\sup_{t>0}
t\,n_{\mathcal A}(t)^{1/p}
\leq
M^{1/p}
\sup_{t>0}
t\,n_{\mathcal B}(t)^{1/p},
\]
which proves \eqref{eq:Lorentz-finite-multiplicity}.
\end{proof}

\begin{lemma}
\label{lem:Dunkl-OSC-ball-oscillation}
Assume that \(N\geq 1\), \(1<p<\infty\) and \(1\leq q\leq\infty\). Fix \(s>2\), \(A>0\) and a closed Weyl chamber \(\Gamma\). Let \(r_s\) be the exponent in Lemma~\ref{lem:Dunkl-mixed-cube-to-Lebesgue-oscillation}. For \(\tau\in G\) and \(f\in OSC^{p,q}(\mathbb R^N)\), set
\[
g_\tau(z):=f(\tau z).
\]
Then
\[
\left\|
\left\{
\operatorname{osc}_{r_s}\bigl(
g_\tau;B_\Gamma(z_I,A\ell(I))
\bigr)
\right\}_{I\in\mathbb D_\Gamma}
\right\|_{\ell^{p,q}}
\lesssim
\|f\|_{OSC^{p,q}(\mathbb R^N)},
\]
The implicit constants are independent of \(f\) and \(\tau\).
\end{lemma}
\begin{proof}
Since \(\tau\) is orthogonal,
\[
\operatorname{osc}_{r_s}\bigl(
g_\tau;B_\Gamma(z_I,A\ell(I))
\bigr)
=
\operatorname{osc}_{r_s}\bigl(
f;\tau B_\Gamma(z_I,A\ell(I))
\bigr).
\]
By Lemma~\ref{Nsystems}, there exist \(\nu(I)\in\{1,\ldots,2^N\}\) and \(J_I\in\mathbb D^{\nu(I)}\) such that
\[
B(\tau z_I,A\ell(I))
\subset J_I
\subset C_NB(\tau z_I,A\ell(I)),
\qquad
\ell(J_I)\simeq\ell(I).
\]
Since
\[
\tau B_\Gamma(z_I,A\ell(I))
\subset
B(\tau z_I,A\ell(I))
\subset J_I
\]
and
\[
\bigl|\tau B_\Gamma(z_I,A\ell(I))\bigr|
=
\bigl|B_\Gamma(z_I,A\ell(I))\bigr|
\simeq
\ell(I)^N
\simeq
|J_I|,
\]
we have
\[
\operatorname{osc}_{r_s}\bigl(
f;\tau B_\Gamma(z_I,A\ell(I))
\bigr)
\lesssim
\operatorname{osc}_{r_s}(f;J_I).
\]

For each \(\nu\in\{1,\ldots,2^N\}\), the map
\[
\{I\in\mathbb D_\Gamma:\nu(I)=\nu\}
\longrightarrow
\mathbb D^\nu,
\qquad
I\longmapsto J_I,
\]
has uniformly bounded multiplicity. That is,
\[
\#\left\{
I\in\mathbb D_\Gamma:
\nu(I)=\nu,\ J_I=J
\right\}
\lesssim 1.
\]
Therefore, by Lemma~\ref{lem:Lorentz-finite-multiplicity},
\begin{align*}
\left\|
\left\{
\operatorname{osc}_{r_s}\bigl(
g_\tau;B_\Gamma(z_I,A\ell(I))
\bigr)
\right\}_{I\in\mathbb D_\Gamma}
\right\|_{\ell^{p,q}}
&\lesssim
\sum_{\nu=1}^{2^N}
\left\|
\left\{
\operatorname{osc}_{r_s}(f;J_I)
\right\}_{\substack{I\in\mathbb D_\Gamma\\ \nu(I)=\nu}}
\right\|_{\ell^{p,q}}\\
&\lesssim
\sum_{\nu=1}^{2^N}
\left\|
\left\{
\operatorname{osc}_{r_s}(f;J)
\right\}_{J\in\mathbb D^\nu}
\right\|_{\ell^{p,q}}=
\|f\|_{OSC^{p,q}(\mathbb R^N)}.
\end{align*}
This completes the proof of Lemma~\ref{lem:Dunkl-OSC-ball-oscillation}.
\end{proof}


\begin{lemma}
\label{lem:Weyl-difference-quotient}
Assume that \(N>1\). For \(\sigma,\tau\in G\), define
\begin{align}\label{defGsigmatau}
\mathcal G_{\sigma,\tau}f(z)
:=
\begin{cases}
\displaystyle
\frac{|f(\tau z)-f(\sigma z)|}
{|\tau z-\sigma z|},
&
\tau z\neq\sigma z,\\[3mm]
0,
&
\tau z=\sigma z.
\end{cases}
\end{align}
Then for every \(f\in\dot W^{1,N}(\mathbb R^N)\),
\begin{equation}
\label{eq:Weyl-difference-quotient}
\|\mathcal G_{\sigma,\tau}f\|_{L^N(\mathbb R^N,dx)}
\lesssim_{N,G}
\|\nabla f\|_{L^N(\mathbb R^N,dx)}.
\end{equation}
The implicit constant is independent of \(\sigma,\tau\in G\).
\end{lemma}
\begin{proof}
Let $\mathcal M$ be the Hardy--Littlewood maximal operator on $\mathbb R^N$. By \cite[Theorem~3]{MR1226425}, after choosing the precise representative of $f$, there exists a null set $E\subset\mathbb R^N$ such that
\[
|f(x)-f(y)|
\lesssim_N
|x-y|
\bigl(
\mathcal M(|\nabla f|)(x)
+
\mathcal M(|\nabla f|)(y)
\bigr),
\qquad x,y\in\mathbb R^N\setminus E.
\]
Since $\sigma$ and $\tau$ are orthogonal, they preserve Lebesgue measure. Hence, for almost every $z\in\mathbb R^N$ with $\tau z\neq\sigma z$,
\[
\mathcal G_{\sigma,\tau}f(z)
\lesssim_N
\mathcal M(|\nabla f|)(\tau z)
+
\mathcal M(|\nabla f|)(\sigma z).
\]
The same estimate holds when $\tau z=\sigma z$ by the definition of $\mathcal G_{\sigma,\tau}f$. Therefore, using again the measure-preserving property of $\sigma$ and $\tau$ and the $L^N$-boundedness of $\mathcal M$, which is valid since $N>1$,
\begin{align*}
\|\mathcal G_{\sigma,\tau}f\|_{L^N(\mathbb R^N)}
&\lesssim_N
\|\mathcal M(|\nabla f|)\circ\tau\|_{L^N}
+
\|\mathcal M(|\nabla f|)\circ\sigma\|_{L^N} \\
&=
2\|\mathcal M(|\nabla f|)\|_{L^N}
\lesssim_N
\|\nabla f\|_{L^N(\mathbb R^N)}.
\end{align*}
The implicit constant is independent of $\sigma,\tau\in G$.
\end{proof}
\begin{lemma}
\label{lem:Dunkl-chamber-jump-supercritical0}
Assume that \(N\geq 1\), \(1<p<\infty\), and
\(1\leq q\leq\infty\). For
\(\sigma,\tau\in G\) and \(I\in\mathbb D_\Gamma\), let
\(\theta_{\sigma,\tau,I}\) be as in \eqref{eq:D-and-theta}. For
\(f\in L^1_{\mathrm{loc}}(\mathbb R^N)\), define
\begin{equation}
\label{eq:chamber-jump-average}
a_{\sigma,\tau,I}(f)
:=
\frac1{|I|}
\int_I
(f(\tau z)-f(\sigma z))\,dz.
\end{equation}
Then for every
\(f\in OSC^{p,q}(\mathbb R^N)\),
\begin{equation}
\label{eq:Dunkl-chamber-jump-supercritical}
\left\|
\left\{
\theta_{\sigma,\tau,I}
a_{\sigma,\tau,I}(f)
\right\}_{I\in\mathbb D_\Gamma}
\right\|_{\ell^{p,q}}
\lesssim
\|f\|_{OSC^{p,q}(\mathbb R^N)}.
\end{equation}
The implicit constant is independent of \(\sigma,\tau\in G\).
\end{lemma}
\begin{proof}
The proof is divided into three cases.

{\bf Case 1.} Suppose that \(N\geq 1\), \(N<p<\infty\), and
\(1\leq q\leq\infty\).

Fix
\(\sigma,\tau\in G\), and choose $r>Np/(p-N)$ so that
\[
1-\frac Nr-\frac Np>0.
\]
 For \(k\geq 0\), let
\[
\mathcal I_k
:=
\left\{
I\in\mathbb D_\Gamma:
2^{-k-1}< \theta_{\sigma,\tau,I}\leq 2^{-k}
\right\}.
\]
Then \(\mathbb D_\Gamma=\bigcup_{k\geq0}\mathcal I_k\) disjointly.

By the construction of \(\mathbb D_\Gamma\) and \eqref{eq:biLip-N},
\[
|I|\simeq\ell(I)^N,
\qquad
|z-z_I|\lesssim\ell(I),\quad z\in I.
\]
Hence, for \(I\in\mathcal I_k\),
\[
\sigma I\cup\tau I
\subset
B_I
:=
B\bigl(\sigma z_I,C2^k\ell(I)\bigr)
\]
with a structural constant \(C\), and
$
|B_I|\simeq 2^{kN}|I|.
$
 Therefore, by the orthogonality of \(\sigma,\tau\) and H\"older's
inequality,
\begin{align*}
|a_{\sigma,\tau,I}(f)|
&\leq
\frac1{|I|}
\int_{\tau I}|f-(f)_{B_I,dx}|\,dx
+
\frac1{|I|}
\int_{\sigma I}|f-(f)_{B_I,dx}|\,dx\\
&\lesssim
\left(\frac{|B_I|}{|I|}\right)^{1/r}
\left(
\frac1{|B_I|}
\int_{B_I}|f-(f)_{B_I,dx}|^r\,dx
\right)^{1/r}.
\end{align*}
Using
\(|B_I|/|I|\simeq2^{kN}\), we obtain
\begin{equation}
\label{eq:jump-by-large-ball-osc}
\theta_{\sigma,\tau,I}|a_{\sigma,\tau,I}(f)|
\lesssim
2^{-k(1-N/r)}
\operatorname{osc}_r(f;B_I),
\qquad I\in\mathcal I_k.
\end{equation}

By Lemma~\ref{Nsystems}, for every \(I\in\mathcal I_k\) there are
\(\nu(I)\in\{1,\ldots,2^N\}\) and \(J_I\in\mathbb D^{\nu(I)}\) such that
\[
B_I\subset J_I\subset C_NB_I.
\]
Consequently,
\[
\ell(J_I)\simeq2^k\ell(I),
\qquad
|J_I|\simeq|B_I|,
\qquad
\operatorname{osc}_r(f;B_I)
\lesssim
\operatorname{osc}_r(f;J_I).
\]
For fixed \(k\) and \(\nu\), we claim that
\[
\sup_{J\in\mathbb D^\nu}
\#\left\{
I\in\mathcal I_k:
\nu(I)=\nu,\ J_I=J
\right\}
\lesssim 2^{kN}.
\]
Indeed, if \(J_I=J\), then
$
\ell(I)\simeq2^{-k}\ell(J).
$
Since both \(I\) and \(J\) have dyadic side-lengths, only finitely many
generations of \(I\), uniformly in \(k\) and \(J\), can occur. For each
such generation, \(B_I\subset J\) implies \(\sigma z_I\in J\). If
\(I=\Phi^{-1}(Q)\) and \(I'=\Phi^{-1}(Q')\) are distinct cubes of the
same generation, then
\[
|c_Q-c_{Q'}|\geq\ell(I).
\]
Since \(c_Q=\Phi(z_I)\), \(c_{Q'}=\Phi(z_{I'})\), and \(\sigma\) is
orthogonal, \eqref{eq:biLip-N} gives
\[
|\sigma z_I-\sigma z_{I'}|
=
|z_I-z_{I'}|
\geq C_\Phi^{-1}|c_Q-c_{Q'}|
\geq C_\Phi^{-1}\ell(I).
\]
Thus, for each fixed generation, the balls
\[
B\left(\sigma z_I,\frac{C_\Phi^{-1}\ell(I)}{2}\right),
\qquad
I\in\mathcal I_k,\quad
\nu(I)=\nu,\quad
J_I=J,\quad
\ell(I)=2^{-j},
\]
are pairwise disjoint. Since their centers belong to \(J\) and
\(\ell(I)\lesssim\ell(J)\), all these balls are contained in a fixed
dilation \(CJ\) of \(J\). Hence,
\[
\#\left\{
I\in\mathcal I_k:
\nu(I)=\nu,\ J_I=J,\ \ell(I)=2^{-j}
\right\}
\ell(I)^N
\lesssim
|CJ|
\lesssim
\ell(J)^N,
\]
and therefore,
\[
\#\left\{
I\in\mathcal I_k:
\nu(I)=\nu,\ J_I=J,\ \ell(I)=2^{-j}
\right\}
\lesssim
\left(\frac{\ell(J)}{\ell(I)}\right)^N
\lesssim
2^{kN}.
\]
Summing over the finitely many possible generations yields
\[
\#\left\{
I\in\mathcal I_k:
\nu(I)=\nu,\ J_I=J
\right\}
\lesssim2^{kN}.
\]

Thus, by Lemma~\ref{lem:Lorentz-finite-multiplicity} and
\eqref{eq:jump-by-large-ball-osc},
\begin{align*}
\left\|
\left\{
\theta_{\sigma,\tau,I}a_{\sigma,\tau,I}(f)
\right\}_{I\in\mathcal I_k}
\right\|_{\ell^{p,q}}
&\lesssim
2^{-k(1-N/r)}
\sum_{\nu=1}^{2^N}
\left\|
\left\{
\operatorname{osc}_r(f;J_I)
\right\}_{\substack{I\in\mathcal I_k\\ \nu(I)=\nu}}
\right\|_{\ell^{p,q}}
\\
&\lesssim
2^{-k(1-N/r-N/p)}
\sum_{\nu=1}^{2^N}
\left\|
\left\{
\operatorname{osc}_r(f;J)
\right\}_{J\in\mathbb D^\nu}
\right\|_{\ell^{p,q}}\\
&\lesssim
2^{-k(1-N/r-N/p)}\|f\|_{OSC^{p,q}(\mathbb R^N)}.
\end{align*}


Therefore,
\begin{align*}
\left\|
\left\{
\theta_{\sigma,\tau,I}a_{\sigma,\tau,I}(f)
\right\}_{I\in\mathbb D_\Gamma}
\right\|_{\ell^{p,q}}
&\lesssim
\sum_{k=0}^\infty
\left\|
\left\{
\theta_{\sigma,\tau,I}a_{\sigma,\tau,I}(f)
\right\}_{I\in\mathcal I_k}
\right\|_{\ell^{p,q}}\\
&\lesssim
\sum_{k=0}^\infty
2^{-k(1-N/r-N/p)}
\|f\|_{OSC^{p,q}(\mathbb R^N)}\\
&\lesssim
\|f\|_{OSC^{p,q}(\mathbb R^N)}.
\end{align*}

{\bf Case 2.} Suppose that $N>1$, $p=N$ and $q=\infty$.

By part \ref{OSC-critical-rigidity-1} of
Lemma~\ref{lem:Euclidean-OSC-critical-rigidity},
\[
\|\nabla f\|_{L^N(\mathbb R^N)}
\simeq
\|f\|_{OSC^{N,\infty}(\mathbb R^N)}.
\]

For
\(I=\Phi^{-1}(Q)\in\mathbb D_\Gamma\) and \(z\in I\), we use the bi-Lipschitz property
\eqref{eq:biLip-N} to see that
\[
|z-z_I|
\lesssim
|\Phi(z)-\Phi(z_I)|
\leq \frac{\sqrt N}{2}\ell(Q)
\lesssim \ell(I).
\]
Since \(\sigma\) and \(\tau\) are orthogonal,
\[
|\tau z-\sigma z|
\leq 2|z-z_I|+|\tau z_I-\sigma z_I|
\lesssim \ell(I)+|\tau z_I-\sigma z_I|.
\]
Hence, by the definition of \(\mathcal G_{\sigma,\tau}f\) (see \eqref{defGsigmatau}) and
\eqref{eq:D-and-theta},
\begin{align*}
\theta_{\sigma,\tau,I}|a_{\sigma,\tau,I}(f)|
&\leq
\frac{\theta_{\sigma,\tau,I}}{|I|}
\int_I |f(\tau z)-f(\sigma z)|\,dz\\
&\lesssim
\frac{\ell(I)}{\ell(I)+|\sigma z_I-\tau z_I|}
\bigl(\ell(I)+|\tau z_I-\sigma z_I|\bigr)
\frac1{|I|}
\int_I\mathcal G_{\sigma,\tau}f(z)\,dz\\
&\lesssim
\ell(I)\frac1{|I|}
\int_I\mathcal G_{\sigma,\tau}f(z)\,dz.
\end{align*}

Set
\[
F_{\sigma,\tau}(u)
:=
\mathcal G_{\sigma,\tau}f(\Phi^{-1}u)\chi_\Omega(u).
\]
By the change of variables \(u=\Phi(z)\), together with
Lemma~\ref{lem:Weyl-difference-quotient},
\[
\|F_{\sigma,\tau}\|_{L^N(\mathbb R^N)}
\lesssim
\|\mathcal G_{\sigma,\tau}f\|_{L^N(\mathbb R^N)}
\lesssim
\|\nabla f\|_{L^N(\mathbb R^N)},
\]
uniformly in \(\sigma,\tau\in G\). Moreover, for
\(I=\Phi^{-1}(Q)\),
\[
\ell(I)\frac1{|I|}
\int_I\mathcal G_{\sigma,\tau}f(z)\,dz
=
\ell(Q)\frac1{|Q|}
\int_QF_{\sigma,\tau}(u)\,du
=
|Q|^{\frac1N-1}\int_QF_{\sigma,\tau}(u)\,du.
\]

By the construction of \(\mathbb D_\Gamma\), every
\(I\in\mathbb D_\Gamma\) is of the form \(I=\Phi^{-1}(Q)\) for a unique
\(Q\in\mathbb D_\Omega^0\subset\mathbb D^0\). Applying
Lemma~\ref{lem:Cwikel-dyadic-distribution} with \(s=N\), we deduce that
\[
\left\|
\left\{
\ell(I)\frac1{|I|}
\int_I\mathcal G_{\sigma,\tau}f(z)\,dz
\right\}_{I\in\mathbb D_\Gamma}
\right\|_{\ell^{N,\infty}}\leq \left\|
\left\{
|Q|^{\frac1N-1}
\int_QF_{\sigma,\tau}(u)\,du
\right\}_{Q\in\mathbb D^0}
\right\|_{\ell^{N,\infty}}
\lesssim
\|F_{\sigma,\tau}\|_{L^N(\mathbb R^N)}.
\]
Combining the preceding estimates, we deduce that
\[
\left\|
\left\{
\theta_{\sigma,\tau,I}a_{\sigma,\tau,I}(f)
\right\}_{I\in\mathbb D_\Gamma}
\right\|_{\ell^{N,\infty}}
\lesssim
\|\nabla f\|_{L^N(\mathbb R^N)}\simeq \|f\|_{OSC^{N,\infty}(\mathbb{R}^N)},
\]
where the final equivalence follows from Lemma \ref{lem:Euclidean-OSC-critical-rigidity}, and where the implicit constant is uniform in \(\sigma,\tau\in G\).

{\bf Case 3.} Suppose that \(N>1\) and either \(1<p<N\), \(1\leq q\leq\infty\), or \(p=N\), \(1\leq q<\infty\).

By parts \ref{OSC-critical-rigidity-2} and \ref{OSC-critical-rigidity-3} of Lemma~\ref{lem:Euclidean-OSC-critical-rigidity}, \(f\) is constant almost everywhere. Hence,
\[
f(\tau z)-f(\sigma z)=0
\qquad\text{for almost every }z\in\mathbb R^N.
\]
Hence, by \eqref{eq:chamber-jump-average},
\(a_{\sigma,\tau,I}(f)=0\) for every \(I\in\mathbb D_\Gamma\). Therefore,
\[
\left\|
\left\{
\theta_{\sigma,\tau,I}a_{\sigma,\tau,I}(f)
\right\}_{I\in\mathbb D_\Gamma}
\right\|_{\ell^{p,q}}
=0.
\]
This completes the proof of Lemma~\ref{lem:Dunkl-chamber-jump-supercritical0}.
\end{proof}

\begin{lemma}
\label{lem:Dunkl-H-OSC-Lorentz}
Assume that \(N\geq 1\), \(1<p<\infty\), and \(1\leq q\leq\infty\).
Fix \(s>2\) and a closed Weyl chamber \(\Gamma\). Let \(r_s\) be the
exponent in Lemma~\ref{lem:Dunkl-mixed-cube-to-Lebesgue-oscillation}.
For $f\in L^1_{\rm loc}(\mathbb{R}^N)$ and \(\sigma,\tau\in G\), define
\begin{equation}
\label{eq:H-sigma-tau}
H_{\sigma,\tau}(f)(z)
:=
f(\tau z)-f(\sigma z),
\qquad z\in\mathbb R^N.
\end{equation}
Then for any \(f\in OSC^{p,q}(\mathbb R^N)\),
\[
\left\|
\left\{
\operatorname{osc}_{r_s}(H_{\sigma,\tau}(f);I)
\right\}_{I\in\mathbb D_\Gamma}
\right\|_{\ell^{p,q}}
\lesssim
\|f\|_{OSC^{p,q}(\mathbb R^N)},
\]
The implicit constant is independent of \(\sigma,\tau\in G\).
\end{lemma}
\begin{proof}
 Since \(\sigma\) and \(\tau\) are orthogonal,
\[
(H_{\sigma,\tau}(f))_{I,dx}
=
(f)_{\tau I,dx}-(f)_{\sigma I,dx},
\]
and hence, by Minkowski's inequality,
\[
\operatorname{osc}_{r_s}(H_{\sigma,\tau}(f);I)
\leq
\operatorname{osc}_{r_s}(f;\tau I)
+
\operatorname{osc}_{r_s}(f;\sigma I).
\]

Fix \(\gamma\in\{\sigma,\tau\}\). By the construction of \(\mathbb D_\Gamma\) and \eqref{eq:biLip-N},
\[
\gamma I\subset B(\gamma z_I,C\ell(I)),
\qquad
|\gamma I|=|I|\simeq\ell(I)^N.
\]
By Lemma~\ref{Nsystems}, for every \(I\in\mathbb D_\Gamma\) there exist
\(\nu_\gamma(I)\in\{1,\ldots,2^N\}\) and
\(J_I^\gamma\in\mathbb D^{\nu_\gamma(I)}\) such that
\[
\gamma I
\subset
B(\gamma z_I,C\ell(I))
\subset
J_I^\gamma
\subset
C_NB(\gamma z_I,C\ell(I)).
\]
Thus,
$
\ell(J_I^\gamma)\simeq\ell(I),$ and $
|J_I^\gamma|\simeq|\gamma I|.
$
This implies that
\[
\operatorname{osc}_{r_s}(f;\gamma I)
\lesssim
\operatorname{osc}_{r_s}(f;J_I^\gamma).
\]

Note that for each \(\nu\in\{1,\ldots,2^N\}\), the map
\[
I\longmapsto J_I^\gamma,
\qquad
\{I\in\mathbb D_\Gamma:\nu_\gamma(I)=\nu\}
\longrightarrow
\mathbb D^\nu,
\]
has uniformly bounded multiplicity. That is,
\[
\sup_{J\in\mathbb D^\nu}
\#\left\{
I\in\mathbb D_\Gamma:
\nu_\gamma(I)=\nu,\ J_I^\gamma=J
\right\}
\lesssim1.
\]
Therefore, by Lemma~\ref{lem:Lorentz-finite-multiplicity},
\begin{align*}
\left\|
\left\{
\operatorname{osc}_{r_s}(f;\gamma I)
\right\}_{I\in\mathbb D_\Gamma}
\right\|_{\ell^{p,q}}
&\lesssim
\sum_{\nu=1}^{2^N}
\left\|
\left\{
\operatorname{osc}_{r_s}(f;J_I^\gamma)
\right\}_{\substack{I\in\mathbb D_\Gamma\\ \nu_\gamma(I)=\nu}}
\right\|_{\ell^{p,q}}\\
&\lesssim
\sum_{\nu=1}^{2^N}
\left\|
\left\{
\operatorname{osc}_{r_s}(f;J)
\right\}_{J\in\mathbb D^\nu}
\right\|_{\ell^{p,q}}
=
\|f\|_{OSC^{p,q}(\mathbb R^N)}.
\end{align*}
Applying this estimate with \(\gamma=\sigma\) and \(\gamma=\tau\), we obtain
\[
\left\|
\left\{
\operatorname{osc}_{r_s}(H_{\sigma,\tau}(f);I)
\right\}_{I\in\mathbb D_\Gamma}
\right\|_{\ell^{p,q}}
\lesssim
\|f\|_{OSC^{p,q}(\mathbb R^N)}.
\]
This completes the proof of Lemma~\ref{lem:Dunkl-H-OSC-Lorentz}.
\end{proof}

\begin{lemma}
\label{lem:Dunkl-commutator-Fourier-NWO-endpoint}
Assume that \(N\geq 1\), \(1<p<\infty\) and \(1\leq q\leq\infty\). Fix $s>2$, \(j\in\{1,\ldots,N\}\), and a closed Weyl chamber \(\Gamma\).  For \(f\in OSC^{p,q}(\mathbb R^N)\), the off-diagonal kernel of
\([R_{{\rm Dunkl},j},M_f]\), denoted by
\(K_{[R_{{\rm Dunkl},j},M_f]}(x,y)\), admits the following factorization:
for almost every \((x,y)\) with \(d(x,y)>0\),
\begin{equation}
\label{eq:Dunkl-commutator-kernel-NWO}
K_{[R_{{\rm Dunkl},j},M_f]}(x,y)
=
\sum_{\varepsilon=0}^3
\sum_{\sigma,\tau\in G}
\sum_{u,v\in\mathbb Z^N}
c_{u,v}
\sum_{I\in\mathbb D_\Gamma}
A_{\sigma,\tau,I,\varepsilon}(f)
E_{u,v,I,\varepsilon}^{\sigma,\tau}(x)
H_{u,v,I,\varepsilon}^{\sigma,\tau}(y),
\end{equation}
where the real numbers
\(
\{c_{u,v}\}_{u,v\in\mathbb Z^N},
\)
coefficients
\(
\left\{
A_{\sigma,\tau,I,\varepsilon}(f)
\right\}_{
\sigma,\tau\in G,\,
I\in\mathbb D_\Gamma,\,
0\leq\varepsilon\leq3},
\)
and functions
\(
E_{u,v,I,\varepsilon}^{\sigma,\tau},
\)
\(
H_{u,v,I,\varepsilon}^{\sigma,\tau}
\)
satisfy
\begin{enumerate}
  \item\label{comm-property1}
  for every \(M_0>2N\),

\left\|

\|f\|_{OSC^{p,q}(\mathbb R^N)}.
\]}
This completes the proof.
\end{proof}

\begin{theorem}
\label{lem:Dunkl-critical-weak-Schatten-upper}
Assume that \(N\geq 1\), \(1<p<\infty\) and \(1\leq q\leq\infty\). Fix \(j\in\{1,\ldots,N\}\). If $f\in OSC^{p,q}(\mathbb{R}^N)$, then
$[R_{{\rm Dunkl},j},M_f]\in S^{p,q}\bigl(L^2(\mathbb R^N,dw)\bigr)$. Moreover,
\begin{equation}
\label{eq:Dunkl-critical-weak-Schatten-upper}
\bigl\|
[R_{{\rm Dunkl},j},M_f]
\bigr\|_
{S^{p,q}(L^2(\mathbb R^N,dw))}
\lesssim
\|f\|_{OSC^{p,q}(\mathbb{R}^N)}.
\end{equation}
\end{theorem}

\begin{proof}
Fix \(s>2\) and a closed Weyl chamber \(\Gamma\). For fixed
\(\varepsilon\in\{0,1,2,3\}\), \(\sigma,\tau\in G\), and
\(u,v\in\mathbb Z^N\), by Lemma~\ref{lem:Dunkl-commutator-Fourier-NWO-endpoint}
\eqref{comm-property2}--\eqref{comm-property3} and the \(G\)-invariance of
\(dw\),
\begin{align*}
\Bigg\{
E_{u,v,I,\varepsilon}^{\sigma,\tau}(\sigma\,\cdot)
\Bigg\}_{I\in\mathbb D_\Gamma},
\qquad
\Bigg\{
\overline{
H_{u,v,I,\varepsilon}^{\sigma,\tau}(\tau\,\cdot)}
\Bigg\}_{I\in\mathbb D_\Gamma}
\end{align*}
are NWO families on \(\Gamma\). Their supports are contained in \(I\) and
\(I^*\), respectively, and \(I^*\) is contained in a fixed enlargement of
\(I\).

For \(\gamma\in G\), define
\begin{align*}
U_\gamma:L^2(\Gamma,dw)&\longrightarrow L^2(\gamma\Gamma,dw),
&
U_\gamma g(x)&:=g(\gamma^{-1}x).
\end{align*}
By the \(G\)-invariance of \(dw\), \(U_\gamma\) is unitary. By
Lemma~\ref{lem:Dunkl-commutator-Fourier-NWO-endpoint}
\eqref{comm-property2}, the operator
\begin{align*}
\sum_{I\in\mathbb D_\Gamma}
A_{\sigma,\tau,I,\varepsilon}(f)
\Big\langle
\,\cdot,\,
\overline{H_{u,v,I,\varepsilon}^{\sigma,\tau}}
\Big\rangle_{L^2(\mathbb R^N,dw)}
E_{u,v,I,\varepsilon}^{\sigma,\tau}
\end{align*}
vanishes on \(L^2(\tau\Gamma,dw)^\perp\), and its range is contained in
\(L^2(\sigma\Gamma,dw)\). Hence, by Lemma~\ref{NWOLEMMAA2} and
Lemma~\ref{lem:Dunkl-commutator-Fourier-NWO-endpoint}
\eqref{comm-property4},
\begin{align*}
&\Bigg\|
\sum_{I\in\mathbb D_\Gamma}
A_{\sigma,\tau,I,\varepsilon}(f)
\Big\langle
\,\cdot,\,
\overline{H_{u,v,I,\varepsilon}^{\sigma,\tau}}
\Big\rangle_{L^2(\mathbb R^N,dw)}
E_{u,v,I,\varepsilon}^{\sigma,\tau}
\Bigg\|_{S^{p,q}(L^2(\mathbb R^N,dw))}
\\
&\qquad=
\Bigg\|
\sum_{I\in\mathbb D_\Gamma}
A_{\sigma,\tau,I,\varepsilon}(f)
\Big\langle
\,\cdot,\,
\overline{
H_{u,v,I,\varepsilon}^{\sigma,\tau}(\tau\,\cdot)}
\Big\rangle_{L^2(\Gamma,dw)}
E_{u,v,I,\varepsilon}^{\sigma,\tau}(\sigma\,\cdot)
\Bigg\|_{S^{p,q}(L^2(\Gamma,dw))}
\\
&\qquad\lesssim
\Bigg\|
\Bigg\{
A_{\sigma,\tau,I,\varepsilon}(f)
\Bigg\}_{I\in\mathbb D_\Gamma}
\Bigg\|_{\ell^{p,q}}
\lesssim
\|f\|_{OSC^{p,q}(\mathbb R^N)}.
\end{align*}

Choose \(M_0>2N\). By
Lemma~\ref{lem:Dunkl-commutator-Fourier-NWO-endpoint}
\eqref{comm-property1},
\begin{align*}
\sum_{u,v\in\mathbb Z^N}|c_{u,v}|<\infty.
\end{align*}
Since \(G\) is finite and \(p>1\), using an equivalent norm on
\(S^{p,q}\) when necessary, the operator series
\begin{align*}
T_f
:=
\sum_{\varepsilon=0}^3
\sum_{\sigma,\tau\in G}
\sum_{u,v\in\mathbb Z^N}
c_{u,v}
\sum_{I\in\mathbb D_\Gamma}
A_{\sigma,\tau,I,\varepsilon}(f)
\Big\langle
\,\cdot,\,
\overline{H_{u,v,I,\varepsilon}^{\sigma,\tau}}
\Big\rangle_{L^2(\mathbb R^N,dw)}
E_{u,v,I,\varepsilon}^{\sigma,\tau}
\end{align*}
converges in \(S^{p,q}(L^2(\mathbb R^N,dw))\), and
\begin{align}
\label{eq:Dunkl-upper-Tf-bound}
\|T_f\|_{S^{p,q}(L^2(\mathbb R^N,dw))}
&\lesssim
\sum_{\varepsilon=0}^3
\sum_{\sigma,\tau\in G}
\sum_{u,v\in\mathbb Z^N}
|c_{u,v}|
\|f\|_{OSC^{p,q}(\mathbb R^N)}
\lesssim
\|f\|_{OSC^{p,q}(\mathbb R^N)}.
\end{align}

It remains to identify \(T_f\) with the commutator. Fix \(r>p\). Since
\(\ell^{p,q}\subset\ell^r\), by
Lemma~\ref{lem:Dunkl-commutator-Fourier-NWO-endpoint}
\eqref{comm-property4},
\begin{align*}
\Bigg\|
\Bigg\{
A_{\sigma,\tau,I,\varepsilon}(f)
\Bigg\}_{I\in\mathbb D_\Gamma}
\Bigg\|_{\ell^r}
\lesssim
\|f\|_{OSC^{p,q}(\mathbb R^N)}.
\end{align*}
For \(m\in\mathbb N\), define
\begin{align*}
T_f^{(m)}
:=
\sum_{\varepsilon=0}^3
\sum_{\sigma,\tau\in G}
\sum_{u,v\in\mathbb Z^N}
c_{u,v}
\sum_{I\in\bigcup_{|k|\leq m}\mathbb D_{\Gamma,k}}
A_{\sigma,\tau,I,\varepsilon}(f)
\Big\langle
\,\cdot,\,
\overline{H_{u,v,I,\varepsilon}^{\sigma,\tau}}
\Big\rangle_{L^2(\mathbb R^N,dw)}
E_{u,v,I,\varepsilon}^{\sigma,\tau}.
\end{align*}
By Lemma~\ref{NWOLEMMAA2}, for fixed
\(\varepsilon,\sigma,\tau,u,v\),
\begin{align*}
&\Bigg\|
\sum_{I\in\bigcup_{|k|>m}\mathbb D_{\Gamma,k}}
A_{\sigma,\tau,I,\varepsilon}(f)
\Big\langle
\,\cdot,\,
\overline{H_{u,v,I,\varepsilon}^{\sigma,\tau}}
\Big\rangle
E_{u,v,I,\varepsilon}^{\sigma,\tau}
\Bigg\|_{S^r}\lesssim
\Bigg\|
\Bigg\{
A_{\sigma,\tau,I,\varepsilon}(f)
\Bigg\}_{I\in\bigcup_{|k|>m}\mathbb D_{\Gamma,k}}
\Bigg\|_{\ell^r}
\longrightarrow0.
\end{align*}
Together with
\(\sum_{u,v}|c_{u,v}|<\infty\) and the preceding uniform
\(\ell^r\)-bound, this implies
\begin{align}
\label{eq:Dunkl-upper-generation-convergence}
\|T_f-T_f^{(m)}\|_{S^r}
\longrightarrow0.
\end{align}

By the independence of the defining exponent of \(OSC^{p,q}\), as in the
proof of Lemma~\ref{lem:Dunkl-commutator-Fourier-NWO-endpoint},
\(f\in L^s_{\mathrm{loc}}(\mathbb R^N,dw)\). Hence, for
\(\varphi,\psi\in C_c^\infty(\mathbb R^N)\),
\(f\varphi,\overline f\,\psi\in L^2(\mathbb R^N,dw)\), and the
commutator form is well defined by
\begin{align*}
\Big\langle
[R_{{\rm Dunkl},j},M_f]\varphi,\psi
\Big\rangle
:=
\Big\langle
R_{{\rm Dunkl},j}(f\varphi),\psi
\Big\rangle
-
\Big\langle
R_{{\rm Dunkl},j}\varphi,\overline f\,\psi
\Big\rangle.
\end{align*}
Let \(R_{{\rm Dunkl},j}^{(m)}\) denote the smooth annular truncation
obtained from the partition \eqref{supportcondi} by restricting the
generations to \(|k|\leq m\). By the construction in the proof of
Lemma~\ref{lem:Dunkl-Riesz-Fourier-NWO-endpoint}, together with
\eqref{keydecomp} and the absolute summability in
Lemma~\ref{lem:Dunkl-commutator-Fourier-NWO-endpoint}
\eqref{comm-property1},
\begin{align}
\label{eq:Dunkl-upper-truncated-identification}
\big\langle T_f^{(m)}\varphi,\psi\big\rangle
=
\Big\langle
R_{{\rm Dunkl},j}^{(m)}(f\varphi),\psi
\Big\rangle
-
\Big\langle
R_{{\rm Dunkl},j}^{(m)}\varphi,\overline f\,\psi
\Big\rangle.
\end{align}
By the \(L^2(dw)\) truncated-kernel representation of the Dunkl Riesz
transform,
\begin{align*}
R_{{\rm Dunkl},j}^{(m)}g
\longrightarrow
R_{{\rm Dunkl},j}g
\qquad\text{in }L^2(\mathbb R^N,dw),
\qquad g\in L^2(\mathbb R^N,dw).
\end{align*}
Therefore, by \eqref{eq:Dunkl-upper-truncated-identification},
\begin{align*}
\lim_{m\to\infty}
\big\langle T_f^{(m)}\varphi,\psi\big\rangle
=
\Big\langle
[R_{{\rm Dunkl},j},M_f]\varphi,\psi
\Big\rangle.
\end{align*}
On the other hand, by
\eqref{eq:Dunkl-upper-generation-convergence},
\begin{align*}
\lim_{m\to\infty}
\big\langle T_f^{(m)}\varphi,\psi\big\rangle
=
\big\langle T_f\varphi,\psi\big\rangle.
\end{align*}
Thus, \(T_f\) is the bounded extension of
\([R_{{\rm Dunkl},j},M_f]\). Combining this identification with
\eqref{eq:Dunkl-upper-Tf-bound}, we obtain
\begin{align*}
\bigl\|
[R_{{\rm Dunkl},j},M_f]
\bigr\|_{S^{p,q}(L^2(\mathbb R^N,dw))}
\lesssim
\|f\|_{OSC^{p,q}(\mathbb R^N)}.
\end{align*}
This completes the proof of Theorem~\ref{lem:Dunkl-critical-weak-Schatten-upper}.
\end{proof}

\begin{lemma}\label{lem:Cwikel-dyadic-distribution}
Let $1<s<\infty$, and let $\mathbb D^0$ be the standard dyadic grid
in $\mathbb R^N$. Suppose that
$
F\in L^s(\mathbb R^N).
$
Then there exists a constant $C>0$ such that
\[
\left\|\left\{|Q|^{-1/s'}\int_Q |F(x)|\,dx\right\}_{Q\in\mathbb D^0}\right\|_{\ell^{s,\infty}}
\leq C
\|F\|_{L^s(\mathbb R^N)}.
\]
\end{lemma}
\begin{proof}
Let $\mathcal E\subset\mathbb D^0$ be finite and set
\[
G_{\mathcal E}(x)
:=
\sum_{Q\in\mathcal E}|Q|^{-1/s'}\chi_Q(x).
\]
Since the dyadic cubes containing any fixed $x$ are totally ordered by inclusion, for each
$x\in\bigcup_{Q\in\mathcal E}Q$ we may choose a cube $Q_x\in\mathcal E$ of minimal volume such that $x\in Q_x$. Every $Q\in\mathcal E$ containing $x$ is then a dyadic ancestor of $Q_x$. Since the volume increases by the factor $2^N$ at each generation,
\[
G_{\mathcal E}(x)
=
\sum_{\substack{Q\in\mathcal E\\ x\in Q}}|Q|^{-1/s'}
\leq
|Q_x|^{-1/s'}
\sum_{k=0}^{\infty}2^{-kN/s'}
\lesssim_{N,s}
\sup_{Q\in\mathcal E}|Q|^{-1/s'}\chi_Q(x).
\]
Consequently,
\begin{align*}
\|G_{\mathcal E}\|_{L^{s'}(\mathbb R^N)}^{s'}
\lesssim_{N,s}
\int_{\mathbb R^N}
\sup_{Q\in\mathcal E}|Q|^{-1}\chi_Q(x)\,dx\leq
\int_{\mathbb R^N}
\sum_{Q\in\mathcal E}|Q|^{-1}\chi_Q(x)\,dx
=
\#\mathcal E.
\end{align*}
and hence,
\[
\|G_{\mathcal E}\|_{L^{s'}(\mathbb R^N)}
\lesssim_{N,s}
(\#\mathcal E)^{1/s'}.
\]
By H\"older's inequality,
\begin{align*}
\sum_{Q\in\mathcal E}|Q|^{-1/s'}\int_Q|F(x)|\,dx
=
\int_{\mathbb R^N}|F(x)|G_{\mathcal E}(x)\,dx\lesssim_{N,s}
\|F\|_{L^s(\mathbb R^N)}(\#\mathcal E)^{1/s'}.
\end{align*}

Fix $t>0$ and let $\mathcal E$ be any finite subcollection of the dyadic cubes satisfying
\[
|Q|^{-1/s'}\int_Q|F(x)|\,dx>t.
\]
Then
\[
t\,\#\mathcal E
<
\sum_{Q\in\mathcal E}|Q|^{-1/s'}\int_Q|F(x)|\,dx
\lesssim_{N,s}
\|F\|_{L^s(\mathbb R^N)}(\#\mathcal E)^{1/s'},
\]
so
\[
\#\mathcal E
\lesssim_{N,s}
t^{-s}\|F\|_{L^s(\mathbb R^N)}^s.
\]
Since $\mathcal E$ is arbitrary,
\[
\#\Bigg\{
Q\in\mathbb D^0:
|Q|^{-1/s'}\int_Q|F(x)|\,dx>t
\Bigg\}
\lesssim_{N,s}
t^{-s}\|F\|_{L^s(\mathbb R^N)}^s,
\qquad t>0.
\]
The asserted weak $\ell^s$ estimate follows. This completes the proof of Lemma~\ref{lem:Cwikel-dyadic-distribution}.
\end{proof}
\begin{lemma}
\label{lem:Euclidean-OSC-critical-rigidity}
Assume that \(N>1\). Then the following statements hold.
\begin{enumerate}
  \item\label{OSC-critical-rigidity-1}
  For every $f\in L^1_{\rm loc}(\mathbb{R}^N)$, we have
  \begin{equation}
  \label{eq:OSC-critical-Sobolev}
  \|\nabla f\|_{L^N(\mathbb R^N)}
  \simeq
  \|f\|_{OSC^{N,\infty}(\mathbb R^N)}.
  \end{equation}
  \item\label{OSC-critical-rigidity-2}
  If \(1\leq q<\infty\) and
  \(f\in OSC^{N,q}(\mathbb R^N)\), then \(f\) is constant almost
  everywhere.
  \item\label{OSC-critical-rigidity-3}
  If \(1\leq p<N\), \(1\leq q\leq\infty\), and
  \(f\in OSC^{p,q}(\mathbb R^N)\), then \(f\) is constant almost
  everywhere.
\end{enumerate}
\end{lemma}
\begin{proof}
This is a well-known result due to Rochberg--Semmes.
\end{proof}
